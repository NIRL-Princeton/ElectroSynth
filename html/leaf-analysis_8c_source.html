54{

66 if( e->y < VSF) e->y = 0.0f;

117//returns proportion of zero crossings within window size (0.0 would be none in window, 1.0 would be all zero crossings)
118Lfloat tZeroCrossingCounter_tick (tZeroCrossingCounter const z, Lfloat input)
119{
120 z->inBuffer[z->position] = input;
121 int futurePosition = ((z->position + 1) % z->currentWindowSize);
122 Lfloat output = 0.0f;
123
124 //add new value to count
125 if ((z->inBuffer[z->position] * z->inBuffer[z->prevPosition]) < 0.0f)
126 {
127 //zero crossing happened, add it to the count array
128 z->countBuffer[z->position] = 1;
129 z->count++;
130 }
131 else
132 {
133 z->countBuffer[z->position] = 0;
134 }
135
136 //remove oldest value from count
137 if (z->countBuffer[futurePosition] > 0)
138 {
139 z->count--;
140 if (z->count < 0)
141 {
142 z->count = 0;
143 }
144 }
145
146 z->prevPosition = z->position;
147 z->position = futurePosition;
148
149 output = z->count * z->invCurrentWindowSize;
150
151 return output;
152}

159 }
160 else
161 {
162 z->currentWindowSize = z->maxWindowSize;
163 }
164 z->invCurrentWindowSize = 1.0f / z->currentWindowSize;
165}

226void tEnvPD_initToPool (tEnvPD* const xpd, int ws, int hs, int bs, tMempool* const mp)
227{
228 _tMempool* m = *mp;
229 _tEnvPD* x = *xpd = (_tEnvPD*) mpool_calloc(sizeof(_tEnvPD), m);
230 x->mempool = m;
231
232 int period = hs, npoints = ws;
233
234 int i;
235
236 if (npoints < 1) npoints = 1024;
237 if (period < 1) period = npoints/2;
238 if (period < npoints / MAXOVERLAP + 1)
239 period = npoints / MAXOVERLAP + 1;
240
241 x->x_npoints = npoints;
242 x->x_phase = 0;
243 x->x_period = period;
244
245 x->windowSize = npoints;
246 x->hopSize = period;
247 x->blockSize = bs;
248
249 for (i = 0; i < MAXOVERLAP; i++) x->x_sumbuf[i] = 0.0f;
250 for (i = 0; i < npoints; i++)
251 x->buf[i] = (1.0f - cosf((2.0f * PI * i) / npoints))/npoints;
252 for (; i < npoints+INITVSTAKEN; i++) x->buf[i] = 0.0f;
253
254 x->x_f = 0;
255
256 x->x_allocforvs = INITVSTAKEN;
257
258 if (x->x_period % x->blockSize)
259 {
260 x->x_realperiod = x->x_period + x->blockSize - (x->x_period % x->blockSize);
261 }
262 else
263 {
264 x->x_realperiod = x->x_period;
265 }
266}

280void tEnvPD_processBlock(tEnvPD const x, Lfloat* in)
281{
282 int n = x->blockSize;
283
284 int count;
285 t_sample *sump;
286 in += n;
287 for (count = x->x_phase, sump = x->x_sumbuf;
288 count < x->x_npoints; count += x->x_realperiod, sump++)
289 {
290 t_sample *hp = x->buf + count;
291 t_sample *fp = in;
292 t_sample sum = *sump;
293 int i;
294
295 for (i = 0; i < n; i++)
296 {
297 fp--;
298 sum += *hp++ * (*fp * *fp);
299 }
300 *sump = sum;
301 }
302 sump[0] = 0.0f;
303 x->x_phase -= n;
304 if (x->x_phase < 0)
305 {
306 x->x_result = x->x_sumbuf[0];
307 for (count = x->x_realperiod, sump = x->x_sumbuf;
308 count < x->x_npoints; count += x->x_realperiod, sump++)
309 sump[0] = sump[1];
310 sump[0] = 0.0f;
311 x->x_phase = x->x_realperiod - n;
312 }
313}

411static void atkdtk_envelope(tAttackDetection const a, Lfloat *in)
412{
413
414 int i = 0;
415 Lfloat tmp;
416 for(i = 0; i < a->blockSize; ++i){
417 tmp = fastabsf(in[i]);
418
419 if(tmp > a->env)
420 a->env = a->atk_coeff * (a->env - tmp) + tmp;
421 else
422 a->env = a->rel_coeff * (a->env - tmp) + tmp;
423 }
424}

543// main analysis function
544static void snac_analyzeframe(tSNAC const s)
545{
546 int n, tindex = s->timeindex;
547 int framesize = s->framesize;
548 int mask = framesize - 1;
549 Lfloat norm = 1.f / sqrtf((Lfloat)(framesize * 2));
550
551 Lfloat *inputbuf = s->inputbuf;
552 Lfloat *processbuf = s->processbuf;
553
554 // copy input to processing buffers
555 for(n=0; n<framesize; n++)
556 {
557 processbuf[n] = inputbuf[tindex] * norm;
558 tindex++;
559 tindex &= mask;
560 }
561
562 // zeropadding
563 for(n=framesize; n<(framesize<<1); n++) processbuf[n] = 0.;
564
565 // call analysis procedures
566 snac_autocorrelation(s);
567 snac_normalize(s);
568 snac_pickpeak(s);
569 snac_periodandfidelity(s);
570}

572static void snac_autocorrelation(tSNAC const s)
573{
574 int n, m;
575 int framesize = s->framesize;
576 int fftsize = framesize * 2;
577 Lfloat *processbuf = s->processbuf;
578 Lfloat *spectrumbuf = s->spectrumbuf;
579
580 REALFFT(fftsize, processbuf);
581
582 // compute power spectrum
583 processbuf[0] *= processbuf[0]; // DC
584 processbuf[framesize] *= processbuf[framesize]; // Nyquist
585
586 for(n=1; n<framesize; n++)
587 {
588 processbuf[n] = processbuf[n] * processbuf[n]
589 + processbuf[fftsize-n] * processbuf[fftsize-n]; // imag coefficients appear reversed
590 processbuf[fftsize-n] = 0.f;
591 }
592
593 // store power spectrum up to SR/4 for possible later use
594 for(m=0; m<(framesize>>1); m++)
595 {
596 spectrumbuf[m] = processbuf[m];
597 }
598
599 // transform power spectrum to autocorrelation function
600 REALIFFT(fftsize, processbuf);
601 return;
602}

604static void snac_normalize (tSNAC const s)
605{
606 int framesize = s->framesize;
607 int framesizeplustimeindex = s->framesize + s->timeindex;
608 int timeindexminusone = s->timeindex - 1;
609 int n, m;
610 int mask = framesize - 1;
611 int seek = framesize * SEEK;
612 Lfloat *inputbuf = s->inputbuf;
613 Lfloat *processbuf= s->processbuf;
614 Lfloat signal1, signal2;
615
616 // minimum RMS implemented as minimum autocorrelation at index 0
617 // functionally equivalent to white noise floor
618 Lfloat rms = s->minrms / sqrtf(1.0f / (Lfloat)framesize);
619 Lfloat minrzero = rms * rms;
620 Lfloat rzero = processbuf[0];
621 if(rzero < minrzero) rzero = minrzero;
622 double normintegral = (double)rzero * 2.;
623
624 // normalize biased autocorrelation function
625 // inputbuf is circular buffer: timeindex may be non-zero when overlap > 1
626 processbuf[0] = 1;
627 for(n=1, m=s->timeindex+1; n<seek; n++, m++)
628 {
629 signal1 = inputbuf[(n + timeindexminusone)&mask];
630 signal2 = inputbuf[(framesizeplustimeindex - n)&mask]; //could this be switched to Lfloat resolution without issue? -JS
631 normintegral -= (double)(signal1 * signal1 + signal2 * signal2);
632 processbuf[n] /= (Lfloat)normintegral * 0.5f;
633 }
634
635 // flush instable function tail
636 for(n = seek; n<framesize; n++) processbuf[n] = 0.;
637 return;
638}

640static void snac_periodandfidelity (tSNAC const s)
641{
642 Lfloat periodlength;
643
644 if(s->periodindex)
645 {
646 periodlength = (Lfloat)s->periodindex +
647 interpolate3phase(s->processbuf, s->periodindex);
648 if(periodlength < 8) periodlength = snac_spectralpeak(s, periodlength);
649 s->periodlength = periodlength;

655// select the peak which most probably represents period length
656static void snac_pickpeak (tSNAC const s)
657{
658 int n, peakindex=0;
659 int seek = s->framesize * SEEK;
660 Lfloat *processbuf= s->processbuf;
661 Lfloat maxvalue = 0.;
662 Lfloat biasedpeak;
663 Lfloat *biasbuf = s->biasbuf;
664
665 // skip main lobe
666 for(n=1; n<seek; n++)
667 {
668 if(processbuf[n] < 0.) break;
669 }
670
671 // find interpolated / biased maximum in SNAC function
672 // interpolation finds the 'real maximum'
673 // biasing favours the first candidate
674 for(; n<seek-1; n++)
675 {
676 if(processbuf[n] >= processbuf[n-1])
677 {
678 if(processbuf[n] > processbuf[n+1]) // we have a local peak
679 {
680 biasedpeak = interpolate3max(processbuf, n) * biasbuf[n];
681
682 if(biasedpeak > maxvalue)
683 {
684 maxvalue = biasedpeak;
685 peakindex = n;
686 }
687 }
688 }
689 }
690 s->periodindex = peakindex;
691 return;
692}

694// verify period length via frequency domain (up till SR/4)
695// frequency domain is more precise than lag domain for period lengths < 8
696// argument 'periodlength' is initial estimation from autocorrelation
697static Lfloat snac_spectralpeak (tSNAC const s, Lfloat periodlength)
698{
699 if(periodlength < 4.0f) return periodlength;
700
701 Lfloat max = 0.;
702 int n, startbin, stopbin, peakbin = 0;
703 int spectrumsize = s->framesize>>1;
704 Lfloat *spectrumbuf = s->spectrumbuf;
705 Lfloat peaklocation = (Lfloat)(s->framesize * 2.0f) / periodlength;
706
707 startbin = (int)(peaklocation * 0.8f + 0.5f);
708 if(startbin < 1) startbin = 1;
709 stopbin = (int)(peaklocation * 1.25f + 0.5f);
710 if(stopbin >= spectrumsize - 1) stopbin = spectrumsize - 1;
711
712 for(n=startbin; n<stopbin; n++)
713 {
714 if(spectrumbuf[n] >= spectrumbuf[n-1])
715 {
716 if(spectrumbuf[n] > spectrumbuf[n+1])
717 {
718 if(spectrumbuf[n] > max)
719 {
720 max = spectrumbuf[n];
721 peakbin = n;
722 }
723 }
724 }
725 }

727 // calculate amplitudes in peak region
728 for(n=(peakbin-1); n<(peakbin+2); n++)
729 {
730 spectrumbuf[n] = sqrtf(spectrumbuf[n]);
731 }
732
733 peaklocation = (Lfloat)peakbin + interpolate3phase(spectrumbuf, peakbin);
734 periodlength = (Lfloat)(s->framesize * 2.0f) / peaklocation;
735
736 return periodlength;
737}

742{

1007int tZeroCrossingCollector_tick(tZeroCrossingCollector const z, Lfloat s)
1008{
1009
1010 // Offset s by half of hysteresis, so that zero cross detection is
1011 // centered on the actual zero.
1012 s += z->_hysteresis * 0.5f;
1013
1014 if (z->_num_edges >= (int)z->_size)
1015 reset(z);
1016
1017 if ((z->_frame == z->_window_size/2) && (z->_num_edges == 0))
1018 reset(z);
1019
1020 update_state(z, s);
1021
1022 if ((++z->_frame >= z->_window_size) && !z->_state)
1023 {
1024 // Remove half the size from _frame, so we can continue seamlessly
1025 z->_frame -= z->_window_size / 2;
1026
1027 // We need at least two rising edges.
1028 if (z->_num_edges > 1)
1029 z->_ready = 1;
1030 else
1031 reset(z);
1032 }
1033
1034 return z->_state;
1035}

1088static inline void update_state(tZeroCrossingCollector const z, Lfloat s)
1089{
1090 if (z->_ready)
1091 {
1092 shift(z, z->_window_size / 2);
1093 z->_ready = 0;
1094 z->_peak = z->_peak_update;
1095 z->_peak_update = 0.0f;
1096 }
1097
1098 if (z->_num_edges >= (int)z->_size)
1099 reset(z);
1100
1101 if (s > 0.0f)
1102 {
1103 if (!z->_state)
1104 {
1105 --z->_pos;
1106 z->_pos &= z->_mask;
1107 tZeroCrossingInfo crossing = z->_info[z->_pos & z->_mask];
1108 crossing->_before_crossing = z->_prev;
1109 crossing->_after_crossing = s;
1110 crossing->_peak = s;
1111 crossing->_leading_edge = (int) z->_frame;
1112 crossing->_trailing_edge = INT_MIN;
1113 crossing->_width = 0.0f;
1114 ++z->_num_edges;
1115 z->_state = 1;
1116 }
1117 else
1118 {
1119 tZeroCrossingInfo_updatePeak(z->_info[z->_pos & z->_mask], s, z->_frame);
1120 }
1121 if (s > z->_peak_update)
1122 {
1123 z->_peak_update = s;
1124 }
1125 }
1126 else if (z->_state && (s < z->_hysteresis))
1127 {
1128 z->_state = 0;
1129 z->_info[z->_pos & z->_mask]->_trailing_edge = z->_frame;
1130 if (z->_peak == 0.0f)
1131 z->_peak = z->_peak_update;
1132 }
1133
1134 if (z->_frame > z->_window_size * 2)
1135 reset(z);
1136
1137 z->_prev = s;
1138}

1140static inline void shift(tZeroCrossingCollector const z, int n)
1141{
1142 tZeroCrossingInfo crossing = z->_info[z->_pos & z->_mask];
1143
1144 crossing->_leading_edge -= n;
1145 if (!z->_state)
1146 crossing->_trailing_edge -= n;
1147 int i = 1;
1148 for (; i != z->_num_edges; ++i)
1149 {
1150 int idx = (z->_pos + i) & z->_mask;
1151 z->_info[idx]->_leading_edge -= n;
1152 int edge = (z->_info[idx]->_trailing_edge -= n);
1153 if (edge < 0.0f)
1154 break;
1155 }
1156 z->_num_edges = i;
1157}

1224void tBitset_setMultiple (tBitset const b, int index, int n, unsigned int val)
1225{
1226 // Check that the index (i) does not get past size
1227 if (index > b->_bit_size)
1228 return;
1229
1230 // Check that the n does not get past the size
1231 if ((index+n) > b->_bit_size)
1232 n = b->_bit_size - index;
1233
1234 // index is the bit index, i is the integer index
1235 int i = index / b->_value_size;
1236
1237 // Do the first partial int
1238 int mod = index & (b->_value_size - 1);
1239 if (mod)
1240 {
1241 // mask off the high n bits we want to set
1242 mod = b->_value_size - mod;
1243
1244 // Calculate the mask
1245 unsigned int mask = ~(UINT_MAX >> mod);
1246
1247 // Adjust the mask if we're not going to reach the end of this int
1248 if (n < mod)
1249 mask &= (UINT_MAX >> (mod - n));
1250
1251 if (val)
1252 b->_bits[i] |= mask;
1253 else
1254 b->_bits[i] &= ~mask;
1255
1256 // Fast exit if we're done here!
1257 if (n < mod)
1258 return;
1259
1260 n -= mod;
1261 ++i;
1262 }
1263
1264 // Write full ints while we can - effectively doing value_size bits at a time
1265 if (n >= b->_value_size)
1266 {
1267 // Store a local value to work with
1268 unsigned int val_ = val ? UINT_MAX : 0;
1269
1270 do
1271 {
1272 b->_bits[i++] = val_;
1273 n -= b->_value_size;
1274 }
1275 while (n >= b->_value_size);
1276 }
1277
1278 // Now do the final partial int, if necessary
1279 if (n)
1280 {
1281 mod = n & (b->_value_size - 1);
1282
1283 // Calculate the mask
1284 unsigned int mask = (1 << mod) - 1;
1285
1286 if (val)
1287 b->_bits[i] |= mask;
1288 else
1289 b->_bits[i] &= ~mask;
1290 }
1291}

1328int tBACF_getCorrelation (tBACF const b, int pos)
1329{
1330 int value_size = b->_bitset->_value_size;
1331 const int index = pos / value_size;
1332 const int shift = pos % value_size;
1333
1334 const unsigned int* p1 = b->_bitset->_bits;
1335 const unsigned int* p2 = b->_bitset->_bits + index;
1336 int count = 0;
1337
1338 if (shift == 0)
1339 {
1340 for (unsigned i = 0; i != b->_mid_array; ++i)
1341 {
1342 // built in compiler popcount functions should be faster but we want this to be portable
1343 // could try to add some define that call the correct function depending on compiler
1344 // or let the user pointer popcount() to whatever they want
1345 // something to look into...
1346#ifdef __GNUC__
1347 count += __builtin_popcount(*p1++ ^ *p2++);
1348#elif _MSC_VER
1349 count += __popcnt(*p1++ ^ *p2++);
1350#else
1351 count += popcount(*p1++ ^ *p2++);
1352#endif
1353 }
1354 }
1355 else
1356 {
1357 const int shift2 = value_size - shift;
1358 for (unsigned i = 0; i != b->_mid_array; ++i)
1359 {
1360 unsigned int v = *p2++ >> shift;
1361 v |= *p2 << shift2;
1362#ifdef __GNUC__
1363 count += __builtin_popcount(*p1++ ^ v++);
1364#elif _MSC_VER
1365 count += __popcnt(*p1++ ^ v++);
1366#else
1367 count += popcount(*p1++ ^ v++);
1368#endif
1369 }
1370 }
1371 return count;
1372}

1494{

1503 {
1504 tZeroCrossingInfo edge2 = tZeroCrossingCollector_getCrossing(p->_zc, i);
1505 if (edge2->_peak >= threshold)

1551static inline void set_bitstream(tPeriodDetector const p)
1552{
1553 Lfloat threshold = tZeroCrossingCollector_getPeak(p->_zc) * PULSE_THRESHOLD;
1554 unsigned int leading_edge = tZeroCrossingCollector_getWindowSize(p->_zc);
1555 unsigned int trailing_edge = 0;
1556
1557 p->_num_pulses = 0;
1558 tBitset_clear(p->_bits);
1559
1560 for (int i = 0; i != tZeroCrossingCollector_getNumEdges(p->_zc); ++i)
1561 {
1562 tZeroCrossingInfo info = tZeroCrossingCollector_getCrossing(p->_zc, i);
1563 if (info->_peak >= threshold)
1564 {
1565 ++p->_num_pulses;
1566 if (info->_leading_edge < leading_edge)
1567 leading_edge = info->_leading_edge;
1568 if (info->_trailing_edge > trailing_edge)
1569 trailing_edge = info->_trailing_edge;
1570 int pos = fmax(info->_leading_edge, 0);
1571 int n = info->_trailing_edge - pos;
1572 tBitset_setMultiple(p->_bits, pos, n, 1);
1573 }
1574 }
1575 p->_half_empty = (leading_edge > p->_mid_point) || (trailing_edge < p->_mid_point);
1576}

1578static inline void autocorrelate(tPeriodDetector const p)
1579{
1580 Lfloat threshold = tZeroCrossingCollector_getPeak(p->_zc) * PULSE_THRESHOLD;
1581
1582 _sub_collector collect;
1583 sub_collector_init(&collect, &p->_zc, p->_periodicity_diff_threshold, p->_range);
1584
1585 if (p->_half_empty || p->_num_pulses < 2)
1586 {
1587 p->_fundamental.periodicity = -1.0f;
1588 return;
1589 }
1590 else
1591 {
1592 int shouldBreak = 0;
1593 int n = tZeroCrossingCollector_getNumEdges(p->_zc);
1594 for (int i = 0; i != n - 1; ++i)
1595 {
1596 tZeroCrossingInfo curr = tZeroCrossingCollector_getCrossing(p->_zc, i);
1597 if (curr->_peak >= threshold)
1598 {
1599 for (int j = i + 1; j != n; ++j)
1600 {
1601 tZeroCrossingInfo next = tZeroCrossingCollector_getCrossing(p->_zc, j);
1602 if (next->_peak >= threshold)
1603 {
1604 int period = tZeroCrossingInfo_period(curr, &next);
1605 if (period > p->_mid_point)
1606 break;
1607 if (period >= p->_min_period)
1608 {
1609
1610 int count = tBACF_getCorrelation(p->_bacf, period);
1611
1612 int mid = p->_bacf->_mid_array * CHAR_BIT * sizeof(unsigned int);
1613
1614 int start = period;
1615
1616 if ((collect._fundamental._period == -1.0f) && count == 0)
1617 {
1618 if (tBACF_getCorrelation(p->_bacf, period / 2.0f) == 0)
1619 count = -1;
1620 }
1621 else if (period < 32) // Search minimum if the resolution is low
1622 {
1623 // Search upwards for the minimum autocorrelation count
1624 for (int d = start + 1; d < mid; ++d)
1625 {
1626 int c = tBACF_getCorrelation(p->_bacf, d);
1627 if (c > count)
1628 break;
1629 count = c;
1630 period = d;
1631 }
1632 // Search downwards for the minimum autocorrelation count
1633 for (int d = start - 1; d > p->_min_period; --d)
1634 {
1635 int c = tBACF_getCorrelation(p->_bacf, d);
1636 if (c > count)
1637 break;
1638 count = c;
1639 period = d;
1640 }
1641 }
1642
1643 if (count == -1)
1644 {
1645 shouldBreak = 1;
1646 break; // Return early if we have false correlation
1647 }
1648 Lfloat periodicity = 1.0f - (count * p->_weight);
1649 _auto_correlation_info info = { i, j, (int) period, periodicity };
1650 sub_collector_process(&collect, info);
1651 if (count == 0)
1652 {
1653 shouldBreak = 1;
1654 break; // Return early if we have perfect correlation
1655 }
1656 }
1657 }
1658 }
1659 }
1660 if (shouldBreak > 0) break;
1661 }
1662 }
1663
1664 // Get the final resuts
1665 sub_collector_get(&collect, collect._fundamental, &p->_fundamental);
1666}

1696static inline int sub_collector_try_sub_harmonic(_sub_collector* collector, int harmonic, _auto_correlation_info info, Lfloat incoming_period)
1697{
1698 if (fabsf(incoming_period - collector->_first_period) < collector->_periodicity_diff_threshold)
1699 {
1700 // If incoming is a different harmonic and has better
1701 // periodicity ...
1702 if (info._periodicity > collector->_fundamental._periodicity &&
1703 harmonic != collector->_fundamental._harmonic)
1704 {
1705 Lfloat periodicity_diff = fabsf(info._periodicity - collector->_fundamental._periodicity);
1706
1707 // If incoming periodicity is within the harmonic
1708 // periodicity threshold, then replace _fundamental with
1709 // incoming. Take note of the harmonic for later.
1710 if (periodicity_diff <= collector->_harmonic_threshold)
1711 {
1712 collector->_fundamental._i1 = info._i1;
1713 collector->_fundamental._i2 = info._i2;
1714 collector->_fundamental._periodicity = info._periodicity;
1715 collector->_fundamental._harmonic = harmonic;
1716 }
1717
1718 // If not, then we save incoming (replacing the current
1719 // _fundamental).
1720 else
1721 {
1722 sub_collector_save(collector, info);
1723 }
1724 }
1725 return 1;
1726 }
1727 return 0;
1728}

1730static inline int sub_collector_process_harmonics(_sub_collector* collector, _auto_correlation_info info)
1731{
1732 if (info._period < collector->_first_period)
1733 return 0;
1734
1735 Lfloat incoming_period = sub_collector_period_of(collector, info);
1736 int multiple = fmaxf(1.0f, roundf( incoming_period / collector->_first_period));
1737 return sub_collector_try_sub_harmonic(collector, fmin(collector->_range, multiple), info, incoming_period/multiple);
1738}

1741{

1744
1745 else if (sub_collector_process_harmonics(collector, info))
1746 return;

1748 else if (info._periodicity > collector->_fundamental._periodicity)
1749 sub_collector_save(collector, info);
1750}

1893static inline void bias(tPitchDetector const p, _pitch_info incoming)
1894{
1895 ++p->_frames_after_shift;
1896 int shifted = 0;
1897
1898 _pitch_info result;
1899
1900 //=============================================================================
1901 //_pitch_info result = bias(current, incoming, shift);
1902 {
1903 Lfloat error = p->_current.frequency * 0.015625; // approx 1/4 semitone
1904 Lfloat diff = fabsf(p->_current.frequency - incoming.frequency);
1905 int done = 0;
1906
1907 // Try fundamental
1908 if (diff < error)
1909 {
1910 result = incoming;
1911 done = 1;
1912 }
1913 // Try harmonics and sub-harmonics
1914 else if (p->_frames_after_shift > 1)
1915 {
1916 if (p->_current.frequency > incoming.frequency)
1917 {
1918 int multiple = roundf(p->_current.frequency / incoming.frequency);
1919 if (multiple > 1)
1920 {
1921 Lfloat f = incoming.frequency * multiple;
1922 if (fabsf(p->_current.frequency - f) < error)
1923 {
1924 result.frequency = f;
1925 result.periodicity = incoming.periodicity;
1926 done = 1;
1927 }
1928 }
1929 }
1930 else
1931 {
1932 int multiple = roundf(incoming.frequency / p->_current.frequency);
1933 if (multiple > 1)
1934 {
1935 Lfloat f = incoming.frequency / multiple;
1936 if (fabsf(p->_current.frequency - f) < error)
1937 {
1938 result.frequency = f;
1939 result.periodicity = incoming.periodicity;
1940 done = 1;
1941 }
1942 }
1943 }
1944 }
1945 // Don't do anything if the latest autocorrelation is not periodic
1946 // enough. Note that we only do this check on frequency shifts (i.e. at
1947 // this point, we are looking at a potential frequency shift, after
1948 // passing through the code above, checking for fundamental and
1949 // harmonic matches).
1950 if (!done)
1951 {
1952 if (p->_pd->_fundamental.periodicity > MIN_PERIODICITY)
1953 {
1954 // Now we have a frequency shift
1955 shifted = 1;
1956 result = incoming;
1957 }
1958 else result = p->_current;
1959 }
1960 }
1961
1962 //=============================================================================
1963
1964 // Don't do anything if incoming is not periodic enough
1965 // Note that we only do this check on frequency shifts
1966 if (shifted)
1967 {
1968 Lfloat periodicity = p->_pd->_fundamental.periodicity;
1969 if (periodicity >= ONSET_PERIODICITY)
1970 {
1971 p->_frames_after_shift = 0;
1972 p->_current = result;
1973 }
1974 else if (periodicity < MIN_PERIODICITY)
1975 {
1976 p->_current.frequency = 0.0f;
1977 p->_current.periodicity = 0.0f;
1978 }
1979 }
1980 else
1981 {
1982 p->_current = result;
1983 }
1984}

2026int tDualPitchDetector_tick (tDualPitchDetector const p, Lfloat sample)
2027{
2028 tPeriodDetection_tick(p->_pd1, sample);
2029 int ready = tPitchDetector_tick(p->_pd2, sample);
2030
2031 if (ready)
2032 {
2033 int pd2_indeterminate = tPitchDetector_indeterminate(p->_pd2);
2034 int disagreement = 0;
2035 Lfloat period = tPeriodDetection_getPeriod(p->_pd1);
2036 if (!pd2_indeterminate && period != 0.0f)
2037 {
2038 _pitch_info _i1;
2039 _i1.frequency = p->sampleRate / tPeriodDetection_getPeriod(p->_pd1);
2040 _i1.periodicity = tPeriodDetection_getFidelity(p->_pd1);
2041 _pitch_info _i2 = p->_pd2->_current;
2042
2043 Lfloat pd1_diff = fabsf(_i1.frequency - p->_mean);
2044 Lfloat pd2_diff = fabsf(_i2.frequency - p->_mean);
2045
2046 _pitch_info i;
2047 disagreement = fabsf(_i1.frequency - _i2.frequency) > (p->_mean * 0.03125f);
2048 // If they agree, we'll use bacf
2049 if (!disagreement) i = _i2;
2050 // A disagreement implies a change
2051 // Start with smaller changes
2052 else if (pd2_diff < p->_mean * 0.03125f) i = _i2;
2053 else if (pd1_diff < p->_mean * 0.03125f) i = _i1;
2054 // Now filter out lower fidelity stuff
2055 else if (_i1.periodicity < p->thresh) return ready;
2056 // Changing up (bacf tends to lead changes)
2057 else if ((_i1.frequency > p->_mean && _i2.frequency > p->_mean) &&
2058 (_i1.frequency < _i2.frequency) &&
2059 (_i2.periodicity > p->thresh))
2060 {
2061 if (roundf(_i2.frequency / _i1.frequency) > 1) i = _i1;
2062 else i = _i2;
2063 }
2064 // Changing down
2065 else if ((_i1.frequency < p->_mean && _i2.frequency < p->_mean) &&
2066 (_i1.frequency > _i2.frequency) &&
2067 (_i2.periodicity > p->thresh))
2068 {
2069 if (roundf(_i1.frequency / _i2.frequency) > 1) i = _i1;
2070 else i = _i2;
2071 }
2072 // A bit of handling for stuff out of bacf range, won't be as solid but better than nothing
2073 else if (_i1.frequency > p->highest)
2074 {
