26// macros TYPELIST_1, TYPELIST_2, ... TYPELIST_50
27// Each takes a number of arguments equal to its numeric suffix
28// The arguments are type names. TYPELIST_NN generates a typelist containing
29// all types passed as arguments, in that order.

354// class template Typelist
355// The building block of typelists of any length
356// Use it through the TYPELIST_NN macros
357// Defines nested types:
358// Head (first element, a non-typelist type by convention)
359// Tail (second element, can be another typelist)
361
362 template <class T, class U>
363 struct Typelist
364 {
365 typedef T Head;
366 typedef U Tail;
367 };
