49// If stack contains note, returns index. Else returns -1;
50int tStack_contains(tStack const ns, uint16_t noteVal)
51{
52 for (int i = 0; i < ns->size; i++)
53 {
54 if (ns->data[i] == noteVal) return i;
55 }
56 return -1;
57}

59void tStack_add(tStack const ns, uint16_t noteVal)
60{
61 uint8_t j;
62
63 int whereToInsert = 0;
64 if (ns->ordered)
65 {
66 for (j = 0; j < ns->size; j++)
67 {
68 if (noteVal > ns->data[j])
69 {
70 if ((noteVal < ns->data[j+1]) || (ns->data[j+1] == -1))
71 {
72 whereToInsert = j+1;
73 break;
74 }
75 }
76 }
77 }
78
79 //first move notes that are already in the stack one position to the right
80 for (j = ns->size; j > whereToInsert; j--)
81 {
82 ns->data[j] = ns->data[(j - 1)];
83 }
84
85 //then, insert the new note into the front of the stack
86 ns->data[whereToInsert] = noteVal;
87
88 ns->size++;
89}

132// Remove noteVal. return 1 if removed, 0 if not
133int tStack_remove(tStack const ns, uint16_t noteVal)
134{
135 uint8_t k;
136 int foundIndex = tStack_contains(ns, noteVal);
137 int removed = 0;
138
139 if (foundIndex >= 0)
140 {
141 for (k = 0; k < (ns->size - foundIndex); k++)
142 {
143 if ((k+foundIndex) >= (ns->capacity - 1))
144 {
145 ns->data[k + foundIndex] = -1;
146 }
147 else
148 {
149 ns->data[k + foundIndex] = ns->data[k + foundIndex + 1];
150 if ((k + foundIndex) == (ns->size - 1))
151 {
152 ns->data[k + foundIndex + 1] = -1;
153 }
154 }
155
156 }
157 // in case it got put on the stack multiple times
158 foundIndex--;
159 ns->size--;
160 removed = 1;
161 }
162
163 return removed;
164}

166// Doesn't change size of data types
167void tStack_setCapacity(tStack const ns, uint16_t cap)
168{
169 if (cap <= 0)
170 ns->capacity = 1;
171 else if (cap <= STACK_SIZE)
172 ns->capacity = cap;
173 else
174 ns->capacity = STACK_SIZE;
175
176 for (int i = cap; i < STACK_SIZE; i++)
177 {
178 if ((int)ns->data[i] != -1)
179 {
180 ns->data[i] = -1;
181 ns->size -= 1;
182 }
183 }
184
185 if (ns->pos >= cap)
186 {
187 ns->pos = 0;
188 }
189}

206// Next item in order of addition to stack. Return 0-31 if there is a next item to move to. Returns -1 otherwise.
207int tStack_next(tStack const ns)
208{
209 int step = 0;
210 if (ns->size != 0) // if there is at least one note in the stack
211 {
212 if (ns->pos > 0) // if you're not at the most recent note (first one), then go backward in the array (moving from earliest to latest)
213 {
214 ns->pos--;
215 }
216 else
217 {
218 ns->pos = (ns->size - 1); // if you are the most recent note, go back to the earliest note in the array
219 }
220
221 step = ns->data[ns->pos];
222 return step;
223 }
224 else
225 {
226 return -1;
227 }
228}

280 poly->firstReceived[i] = 0;

340int tPoly_noteOn(tPoly const poly, int note, uint8_t vel)
341{
342 // if not in keymap or already on stack, dont do anything. else, add that note.
343 if (tStack_contains(poly->stack, note) >= 0) return -1;
344 else
345 {
346 tPoly_orderedAddToStack(poly, note);
347 tStack_add(poly->stack, note);
348
349 int alteredVoice = -1;
350 int found = 0;
351 for (int i = 0; i < poly->numVoices; i++)
352 {
353 if (poly->voices[i][0] < 0) // if inactive voice, give this note to voice
354 {
355 if (!poly->firstReceived[i] || !poly->pitchGlideIsActive)
356 {
357 tRamp_setVal(poly->ramps[i], note);
358 poly->firstReceived[i] = 1;
359 }
360
361 found = 1;
362
363 poly->voices[i][0] = note;
364 poly->voices[i][1] = vel;
365 poly->lastVoiceToChange = i;
366 poly->notes[note][0] = vel;
367 poly->notes[note][1] = i;
368
369 tRamp_setDest(poly->ramps[i], poly->voices[i][0]);
370
371 alteredVoice = i;
372 break;
373 }
374 }
375
376 if (!found) //steal
377 {
378 int whichVoice, whichNote;
379 for (int j = tStack_getSize(poly->stack) - 1; j >= 0; j--)
380 {
381 whichNote = tStack_get(poly->stack, j);
382 whichVoice = poly->notes[whichNote][1];
383 if (whichVoice >= 0)
384 {
385 poly->lastVoiceToChange = whichVoice;
386 int oldNote = poly->voices[whichVoice][0];
387 poly->voices[whichVoice][0] = note;
388 poly->voices[whichVoice][1] = vel;
389 poly->notes[oldNote][1] = -1; //mark the stolen voice as inactive (in the second dimension of the notes array)
390 poly->notes[note][0] = vel;
391 poly->notes[note][1] = whichVoice;
392
393 if (poly->pitchGlideIsActive)
394 {
395 tRamp_setTime(poly->ramps[whichVoice], poly->glideTime);
396 }
397 else
398 {
399 tRamp_setVal(poly->ramps[whichVoice], note);
400 }
401 tRamp_setDest(poly->ramps[whichVoice], poly->voices[whichVoice][0]);
402
403 alteredVoice = whichVoice;
404
405 break;
406 }
407 }
408 }
409 return alteredVoice;
410 }
411}

474void tPoly_orderedAddToStack(tPoly const poly, uint8_t noteVal)
475{
476 uint8_t j;
477 int myPitch, thisPitch, nextPitch;
478
479 tStack ns = poly->orderStack;
480
481 int whereToInsert = 0;
482
483 for (j = 0; j < ns->size; j++)
484 {
485 myPitch = noteVal;
486 thisPitch = ns->data[j];
487 nextPitch = ns->data[j+1];
488
489 if (myPitch > thisPitch)
490 {
491 if ((myPitch < nextPitch) || (nextPitch == -1))
492 {
493 whereToInsert = j+1;
494 break;
495 }
496 }
497 }
498
499 //first move notes that are already in the stack one position to the right
500 for (j = ns->size; j > whereToInsert; j--)
501 {
502 ns->data[j] = ns->data[(j - 1)];
503 }
504
505 //then, insert the new note into the front of the stack
506 ns->data[whereToInsert] = noteVal;
507
508 ns->size++;
509
510}

630int tSimplePoly_noteOn(tSimplePoly const poly, int note, uint8_t vel)
631{
632 int whichVoice, whichNote, oldNote, alteredVoice;
633 // if not in keymap or already on stack, dont do anything. else, add that note.
634 if (tStack_contains(poly->stack, note) >= 0) return -1;
635 else
636 {
637 alteredVoice = -1;
638 int found = 0;
639 for (int i = 0; i < poly->numVoices; i++)
640 {
641 if (poly->voices[i][0] == -1) // if inactive voice, give this note to voice
642 {
643
644 found = 1;
645
646 poly->voices[i][0] = note;
647 poly->voices[i][1] = vel;
648 poly->notes[note][0] = i;
649 poly->notes[note][1] = vel;
650 poly->voices[i][2] = note; // voices[i][2] is the output midi note, (avoiding the -1 when a voice is inactive)
651
652 alteredVoice = i;
653 tStack_add(poly->stack, note);
654 break;
655 }
656 }
657 if (!found)
658 {
659 //second preference is grabbing one that is in release phase but not finished sounding yet
660 for (int i = 0 ; i < poly->numVoices; i++)
661 {
662 if (poly->voices[i][0] == -2) // if voice is released but still sounding, take over this voice
663 {
664
665 found = 1;
666
667 poly->voices[i][0] = note;
668 poly->voices[i][1] = vel;
669 poly->notes[note][0] = i;
670 poly->notes[note][1] = vel;
671 poly->voices[i][2] = note; // voices[i][2] is the output midi note, (avoiding the -1 when a voice is inactive)
672
673 alteredVoice = i;
674 tStack_add(poly->stack, note);
675 break;
676 }
677 }
678 }
679 if ((!found) && (poly->stealing_on)) //steal
680 {
681 for (int j = tStack_getSize(poly->stack) - 1; j >= 0; j--)
682 {
683 whichNote = tStack_get(poly->stack, j);
684 whichVoice = poly->notes[whichNote][0];
685 if (whichVoice >= 0)
686 {
687 oldNote = poly->voices[whichVoice][0];
688 poly->voices[whichVoice][0] = note;
689 poly->voices[whichVoice][1] = vel;
690 poly->notes[oldNote][0] = -3; //mark the stolen voice as stolen (in the second dimension of the notes array)
691 poly->notes[note][0] = whichVoice;
692 poly->notes[note][1] = vel;
693
694 poly->voices[whichVoice][2] = note;
695
696 alteredVoice = whichVoice;
697 tStack_add(poly->stack, note);
698 break;
699 }
700 }
701 }
702 return alteredVoice;
703 }
704}

707int tSimplePoly_noteOff(tSimplePoly const poly, uint8_t note)
708{
709 int16_t noteToTest = -1;
710
711 tStack_remove(poly->stack, note);
712 poly->notes[note][0] = -1;
713
714 int deactivatedVoice = -1;
715 for (int i = 0; i < poly->maxNumVoices; i++)
716 {
717 if (poly->voices[i][0] == note)
718 {
719 poly->voices[i][0] = -1;
720 poly->voices[i][1] = 0;
721 deactivatedVoice = i;
722 break;
723 }
724 }
725
726 if (poly->recover_stolen)
727 {
728 //grab old notes off the stack if there are notes waiting to replace the free voice
729 if (deactivatedVoice >= 0)
730 {
731 for (int j = 0; j < tStack_getSize(poly->stack); ++j)
732 {
733 noteToTest = tStack_get(poly->stack, j);
734

737 poly->voices[deactivatedVoice][0] = noteToTest; //set the newly free voice to use the old stolen note
738 poly->voices[deactivatedVoice][1] = poly->notes[noteToTest][1]; // set the velocity of the voice to be the velocity of that note
739 poly->voices[deactivatedVoice][2] = noteToTest;
740 poly->notes[noteToTest][0] = deactivatedVoice; //mark that it is no longer stolen and is now active
741 return -1;
742 }
743 }
744 }
745 }
746 return deactivatedVoice;

750void tSimplePoly_deactivateVoice(tSimplePoly const poly, uint8_t voice)
751{
752 if (poly->voices[voice][0] == -2) //only do this if the voice is waiting for deactivation (not already reassigned while waiting)
753 {
754 poly->voices[voice][0] = -1;
755 poly->voices[voice][1] = 0;
756 if (poly->recover_stolen)
757 {
758 //grab old notes off the stack if there are notes waiting to replace the free voice
759 for (int j = 0; j < tStack_getSize(poly->stack); ++j)
760 {
761 noteToTest = tStack_get(poly->stack, j); //note to check if it is waiting to be recovered
762
763 if (poly->notes[noteToTest][0] == -3) //if there is a stolen note waiting (marked inactive but on the stack)
764 {
765 poly->voices[voice][0] = noteToTest; //set the newly free voice to use the old stolen note
766 poly->voices[voice][1] = poly->notes[noteToTest][1]; // set the velocity of the voice to be the velocity of that note
