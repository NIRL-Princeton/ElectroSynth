12template <class Base>
13class Factory {
14public:
15 using CreateFunction = std::function<std::unique_ptr<Base>(std::any)>;
16
17 template <typename T, typename... Args>
18 void registerType(const std::string& typeName) {
19 creators[typeName] = [](std::any args) -> std::unique_ptr<Base> {
20 try {
21 auto tupleArgs = std::any_cast<std::tuple<Args...>>(args); // Unpack std::any into tuple
22 return std::apply([](auto&&... unpackedArgs) {
23 return std::make_unique<T>(std::forward<decltype(unpackedArgs)>(unpackedArgs)...); // Create shared_ptr with forwarded arguments
24 }, tupleArgs); // Apply the arguments
25 } catch (const std::bad_any_cast& e) {
26 std::cerr << "std::bad_any_cast: " << e.what() << " (expected tuple)" << std::endl;
27 return nullptr;
28 }
29 };
30 }
31
32 // Create object with arguments wrapped in std::any
33 std::unique_ptr<Base> create(const std::string& typeName, std::any args) const {
34 auto it = creators.find(typeName);
35 if (it != creators.end()) {
36 return std::move(it->second(args)); // Call the creation function with arguments
37 }
38 return nullptr; // Type not found
39 }
40
41private:
42 std::map<std::string, CreateFunction> creators;
43};
