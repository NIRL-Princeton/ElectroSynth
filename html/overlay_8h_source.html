24class OverlayBackgroundRenderer : public OpenGlComponent {
25 public:
26 static constexpr int kNumVertices = 4;
27 static constexpr int kNumFloatsPerVertex = 2;
28 static constexpr int kTotalFloats = kNumVertices * kNumFloatsPerVertex;
29 static constexpr int kIndices = 6;
30
31 OverlayBackgroundRenderer() {
32 shader_ = nullptr;
33 additive_blending_ = false;
34
35 color_ = Colours::black;
36 data_[0] = -1.0f;
37 data_[1] = -1.0f;
38 data_[2] = -1.0f;
39 data_[3] = 1.0f;
40 data_[4] = 1.0f;
41 data_[5] = -1.0f;
42 data_[6] = 1.0f;
43 data_[7] = 1.0f;
44
45 indices_[0] = 0;
46 indices_[1] = 1;
47 indices_[2] = 2;
48 indices_[3] = 1;
49 indices_[4] = 2;
50 indices_[5] = 3;
51
52 setInterceptsMouseClicks(false, false);
53 }
54
55 virtual ~OverlayBackgroundRenderer() { }
56
57 virtual void init(OpenGlWrapper& open_gl) override {
58 open_gl.context.extensions.glGenBuffers(1, &data_buffer_);
59 open_gl.context.extensions.glBindBuffer(juce::gl::GL_ARRAY_BUFFER, data_buffer_);
60
61 GLsizeiptr vert_size = static_cast<GLsizeiptr>(kTotalFloats * sizeof(float));
62 open_gl.context.extensions.glBufferData(juce::gl::GL_ARRAY_BUFFER, vert_size, data_, juce::gl::GL_STATIC_DRAW);
63
64 open_gl.context.extensions.glGenBuffers(1, &indices_buffer_);
65 open_gl.context.extensions.glBindBuffer(juce::gl::GL_ELEMENT_ARRAY_BUFFER, indices_buffer_);
66
67 GLsizeiptr bar_size = static_cast<GLsizeiptr>(kIndices * sizeof(int));
68 open_gl.context.extensions.glBufferData(juce::gl::GL_ELEMENT_ARRAY_BUFFER, bar_size, indices_, juce::gl::GL_STATIC_DRAW);
69
70 shader_ = open_gl.shaders->getShaderProgram(Shaders::kPassthroughVertex, Shaders::kColorFragment);
71 shader_->use();
72 color_uniform_ = getUniform(open_gl, *shader_, "color");
73 position_ = getAttribute(open_gl, *shader_, "position");
74 }
75
76 virtual void render(OpenGlWrapper& open_gl, bool animate) override {
77 drawOverlay(open_gl);
78 }
79
80 void paintBackground(Graphics& g) override { }
81
82 virtual void destroy(juce::OpenGLContext& open_gl) override {
83 shader_ = nullptr;
84 position_ = nullptr;
85 color_uniform_ = nullptr;
86 open_gl.extensions.glDeleteBuffers(1, &data_buffer_);
87 open_gl.extensions.glDeleteBuffers(1, &indices_buffer_);
88
89 data_buffer_ = 0;
90 indices_buffer_ = 0;
91 }
92
93 void setColor(const Colour& color) { color_ = color; }
94
95 force_inline void setAdditiveBlending(bool additive_blending) { additive_blending_ = additive_blending; }
96
97 protected:
98 void drawOverlay(OpenGlWrapper& open_gl) {
99 if (!setViewPort(open_gl))
100 return;
101
102 if (shader_ == nullptr)
103 init(open_gl);
104
105 juce::gl::glEnable(juce::gl::GL_BLEND);
106 juce::gl::glEnable(juce::gl::GL_SCISSOR_TEST);
107 if (additive_blending_)
108 juce::gl::glBlendFunc(juce::gl::GL_SRC_ALPHA, juce::gl::GL_ONE);
109 else
110 juce::gl::glBlendFunc(juce::gl::GL_SRC_ALPHA, juce::gl::GL_ONE_MINUS_SRC_ALPHA);
111
112 shader_->use();
113 color_uniform_->set(color_.getFloatRed(), color_.getFloatGreen(),
114 color_.getFloatBlue(), color_.getFloatAlpha());
115
116 open_gl.context.extensions.glBindBuffer(juce::gl::GL_ARRAY_BUFFER, data_buffer_);
117 open_gl.context.extensions.glBindBuffer(juce::gl::GL_ELEMENT_ARRAY_BUFFER, indices_buffer_);
118
119 open_gl.context.extensions.glVertexAttribPointer(position_->attributeID, kNumFloatsPerVertex, juce::gl::GL_FLOAT,
120 juce::gl::GL_FALSE, kNumFloatsPerVertex * sizeof(float), nullptr);
121 open_gl.context.extensions.glEnableVertexAttribArray(position_->attributeID);
122
123 juce::gl::glDrawElements(juce::gl::GL_TRIANGLES, kIndices, juce::gl::GL_UNSIGNED_INT, nullptr);
124
125 open_gl.context.extensions.glDisableVertexAttribArray(position_->attributeID);
126 open_gl.context.extensions.glBindBuffer(juce::gl::GL_ARRAY_BUFFER, 0);
127 open_gl.context.extensions.glBindBuffer(juce::gl::GL_ELEMENT_ARRAY_BUFFER, 0);
128
129 juce::gl::glDisable(juce::gl::GL_BLEND);
130 juce::gl::glDisable(juce::gl::GL_SCISSOR_TEST);
131 }
132
133 OpenGLShaderProgram* shader_;
134 std::unique_ptr<OpenGLShaderProgram::Uniform> color_uniform_;
135 std::unique_ptr<OpenGLShaderProgram::Attribute> position_;
136
137 Colour color_;
138 bool additive_blending_;
139
140 float data_[kTotalFloats];
141 int indices_[kIndices];
142 GLuint data_buffer_;
143 GLuint indices_buffer_;
144
145 JUCE_DECLARE_NON_COPYABLE_WITH_LEAK_DETECTOR(OverlayBackgroundRenderer)
146};

161 Overlay(String name) : SynthSection(name), size_ratio_(1.0f) , background_(new OverlayBackgroundRenderer()){
162 setSkinOverride(Skin::kOverlay);
163 addOpenGlComponent(background_);
164 }
165 virtual ~Overlay() { }
166
167 void setVisible(bool should_be_visible) override {
168 for (Listener* listener : listeners_) {
169 if (should_be_visible)
170 listener->overlayShown(this);
171 else
172 listener->overlayHidden(this);
173 }
174 Component::setVisible(should_be_visible);
175 }
176
177 virtual void resized() override {
178 background_->setColor(findColour(Skin::kOverlayScreen, true));
179 background_->setBounds(getLocalBounds());
180 }
181
182 virtual void paintBackground(Graphics& g) override { paintOpenGlChildrenBackgrounds(g); }
183
184 void addOverlayListener(Listener* listener) { listeners_.insert(listener); }
185 void removeOverlayListener(Listener* listener) { listeners_.erase(listener); }
186 void setSizeRatio(float ratio) override { size_ratio_ = ratio; }
187
188 protected:
189 float size_ratio_;
190 std::set<Listener*> listeners_;
191 std::shared_ptr<OverlayBackgroundRenderer> background_;
192
193 JUCE_DECLARE_NON_COPYABLE_WITH_LEAK_DETECTOR(Overlay)
194};
