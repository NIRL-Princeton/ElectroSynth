52void glottis_setup_waveform(glottis const glot)
53{
54 Lfloat Rd;
55 Lfloat Ra;
56 Lfloat Rk;
57 Lfloat Rg;
58
59 Lfloat Ta;
60 Lfloat Tp;
61 Lfloat Te;
62
63 Lfloat epsilon;
64 Lfloat shift;
65 Lfloat delta;
66 Lfloat rhs_integral;
67
68 Lfloat lower_integral;
69 Lfloat upper_integral;
70
71 Lfloat omega;
72 Lfloat s;
73 Lfloat y;
74 Lfloat z;
75
76 Lfloat alpha;
77 Lfloat E0;
78
79
80 glot->Rd = 3.0f * (1.0f - glot->tenseness);
81 glot->waveform_length = 1.0f / glot->freq;
82
83 Rd = glot->Rd;
84 if(Rd < 0.5f) Rd = 0.5f;
85 if(Rd > 2.7f) Rd = 2.7f;
86
87
88 Ra = -0.01f + 0.048f*Rd;
89 Rk = 0.224f + 0.118f*Rd;
90 Rg = (Rk*0.25f)*(0.5f + 1.2f*Rk)/(0.11f*Rd-Ra*(0.5f+1.2f*Rk));
91
92
93 Ta = Ra;
94 Tp = (Lfloat)1.0f / (2.0f*Rg);
95 Te = Tp + Tp*Rk;
96
97
98
99 epsilon = (Lfloat)1.0f / Ta;
100 shift = fasterexpf(-epsilon * (1.0f - Te));
101 delta = 1.0f - shift;
102
103
104
105 rhs_integral = (Lfloat)(1.0f/epsilon) * (shift-1.0f) + (1.0f-Te)*shift;
106 rhs_integral = rhs_integral / delta;
107 lower_integral = - (Te - Tp) / 2.0f + rhs_integral;
108 upper_integral = -lower_integral;
109
110
111 omega = M_PI / Tp;
112 s = fast_sinf2(omega * Te);
113
114 y = -M_PI * s * upper_integral / (Tp*2.0f);
115 z = my_faster_logf(y);
116 alpha = z / (Tp/2.0f - Te);
117 E0 = -1.0f / (s * fastExp4(alpha*Te));
118
119
120
121 glot->alpha = alpha;
122 glot->E0 = E0;
123 glot->epsilon = epsilon;
124 glot->shift = shift;
125 glot->delta = delta;
126 glot->Te = Te;
127 glot->omega = omega;
128}

156Lfloat glottis_compute(glottis const glot)
157{
158 LEAF* leaf = glot->mempool->leaf;
159 Lfloat out;
160 Lfloat aspiration;
161 Lfloat noise;
162 Lfloat t;
163 Lfloat intensity;
164
165 out = 0.0f;
166 intensity = 1.0f;
167 glot->time_in_waveform += glot->T;
168
169 if(glot->time_in_waveform > glot->waveform_length) {
170 glot->time_in_waveform -= glot->waveform_length;
171 glottis_setup_waveform(glot);
172
173 }
174
175 t = (glot->time_in_waveform * glot->freq);//multiplying by freq same as divide by length since length is 1/f
176
177 if(t > glot->Te) {
178 out = (-fastExp4(-glot->epsilon * (t-glot->Te)) + glot->shift) / glot->delta;
179 } else {
180#ifdef ARM_MATH_CM7
181 out = glot->E0 * fastExp4(glot->alpha * t) * arm_sin_f32(glot->omega * t);
182#else
183 out = glot->E0 * fastExp4(glot->alpha * t) * sinf(glot->omega * t);
184#endif
185
186 }
187
188 noise = (2.0f * leaf->random()) - 1.0f;
189
190#ifdef ARM_MATH_CM7
191 Lfloat sqr = 0.0f;
192 arm_sqrt_f32(glot->tenseness, &sqr);
193 aspiration = intensity * (1.0f - sqr) * 0.3f * noise;
194#else
195 aspiration = intensity * (1.0f - sqrtf(glot->tenseness)) * 0.3f * noise;
196#endif
197
198 aspiration *= 0.2f;
199
200 out += aspiration;
201
202 return out;
203}

336void tract_calculate_reflections(tract const tr)
337{
338 int i;
339 Lfloat sum;
340
341 for(i = 0; i < tr->n; i++)
342 {
343 Lfloat scaledDiameter = tr->diameter[i] * tr->diameterScale;
344 tr->A[i] = scaledDiameter * scaledDiameter;
345 /* Calculate area from diameter squared*/
346 }
347
348 for(i = 1; i < tr->n; i++)
349 {
350 tr->reflection[i] = tr->new_reflection[i];
351
352 Lfloat divisorTest = (tr->A[i - 1] + tr->A[i]);
353
354 if(tr->A[i] <= 0.0000001f)
355 {
356 tr->new_reflection[i] = 0.99f; /* to prevent bad behavior if 0 */
357 }
358 else
359 {
360 if ((divisorTest >= 0.0000001f)|| (divisorTest <= -0.0000001f))
361 {
362 tr->new_reflection[i] = (tr->A[i - 1] - tr->A[i]) / divisorTest;
363 }
364 }
365 }
366
367 tr->reflection_left = tr->new_reflection_left;
368 tr->reflection_right = tr->new_reflection_right;
369 tr->reflection_nose = tr->new_reflection_nose;
370
371 sum = tr->A[tr->nose_start] + tr->A[tr->nose_start + 1] + tr->noseA[0];
372 if ((sum <= 0.001f) && (sum >= -0.001f))
373 {
374 sum = 0.001f;
375 }
376 Lfloat invSum = 1.0f / sum;
377 tr->new_reflection_left = (Lfloat)(2.0f * tr->A[tr->nose_start] - sum) * invSum;
378 tr->new_reflection_right = (Lfloat)(2.0f * tr->A[tr->nose_start + 1] - sum) * invSum;
379 tr->new_reflection_nose = (Lfloat)(2.0f * tr->noseA[0] - sum) * invSum;
380}

542void tract_addTurbulenceNoiseAtPosition(tract const tr, Lfloat turbulenceNoise, Lfloat position, Lfloat diameter)
543{
544 int i = (int)floorf(position);
545 Lfloat delta = position - i;
546 //Lfloat mapped = LEAF_map(diameter, 0.38f, 0.245f, 0.0f, 1.0f);

549 Lfloat noise0 = turbulenceNoise * (1.0f - delta) * thinness0 * openness * 0.5f;
550 Lfloat noise1 = turbulenceNoise * delta * thinness0 * openness * 0.5f;
551
552 if ((i + 1) < tr->n) {
553 tr->R[i + 1] += noise0;
554 tr->L[i + 1] += noise0;
555 }
556 if ((i + 2) < tr->n) {
557 tr->R[i + 2] += noise1;
558 tr->L[i + 2] += noise1;
559 }

563void tract_compute(tract const tr, Lfloat in, Lfloat lambda)
564{
565 Lfloat r, w;
566 int i;
567 //Lfloat amp;
568 //int current_size;
569 // _transient_pool *thepool;

580 tr->L[n->position] += amp * 0.5f;
581 tr->R[n->position] += amp * 0.5f;

595 tract_addTurbulenceNoise(tr);
