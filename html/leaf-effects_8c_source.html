89void tTalkbox_update(tTalkbox const v)
90{
91 Lfloat fs = v->sampleRate;
92// if(fs < 8000.0f) fs = 8000.0f;
93// if(fs > 96000.0f) fs = 96000.0f;
94
95 int32_t n = (int32_t)(0.01633f * fs); //this sets the window time to 16ms if the buffer is large enough. Buffer needs to be at least 784 samples at 48000
96 if(n > v->bufsize) n = v->bufsize;
97
98 //O = (VstInt32)(0.0005f * fs);
99 v->O = (int32_t)((0.0001f + 0.0004f * v->param[3]) * fs);
100 if (v->O >= ORD_MAX)
101 {
102 v->O = ORD_MAX-1;
103 }
104
105 if(n != v->N) //recalc hanning window
106 {
107 v->N = n;
108 Lfloat dp = TWO_PI / v->N;
109 Lfloat p = 0.0f;
110 for(n=0; n<v->N; n++)
111 {
112 v->window[n] = 0.5f - 0.5f * cosf(p);
113 p += dp;
114 }
115 }
116 v->wet = 0.5f * v->param[0] * v->param[0];
117 v->dry = 2.0f * v->param[1] * v->param[1];
118}

138// warped autocorrelation adapted from ten.enegatum@liam's post on music-dsp 2004-04-07 09:37:51
139//find the order-P autocorrelation array, R, for the sequence x of length L and warping of lambda
140//wAutocorrelate(&pfSrc[stIndex],siglen,R,P,0);
141void tTalkbox_warpedAutocorrelate(Lfloat * x, double* dl, double* Rt, unsigned int L, Lfloat * R, unsigned int P, Lfloat lambda)
142{
143 double r1,r2,r1t;
144 R[0]=0;
145 Rt[0]=0;
146 r1=0;
147 r2=0;
148 r1t=0;
149 for(uint32_t m=0; m<L;m++)
150 {
151 Rt[0] += (double)(x[m]) * (double)(x[m]);
152
153 dl[m]= r1 - (double)(lambda) * (double)(x[m]-r2);
154 r1 = x[m];
155 r2 = dl[m];
156 }
157 for(uint32_t i=1; i<=P; i++)
158 {
159 Rt[i]=0;
160 r1=0;
161 r2=0;
162 for(unsigned int m=0; m<L;m++)
163 {
164 Rt[i] += (double) (dl[m]) * (double)(x[m]);
165
166 r1t = dl[m];
167 dl[m]= r1 - (double)(lambda) * (double)(r1t-r2);
168 r1 = r1t;
169 r2 = dl[m];
170 }
171 }
172 for(uint32_t i=0; i<=P; i++)
173 {
174 R[i]=(Lfloat)(Rt[i]);
175 }
176
177}

179void tTalkbox_lpcDurbin(Lfloat *r, int p, Lfloat *k, Lfloat *g)
180{
181 int i, j;
182 Lfloat a[ORD_MAX], at[ORD_MAX], e=r[0];
183
184 for(i=0; i<=p; i++)
185 {
186 a[i] = 0.0f; //probably don't need to clear at[] or k[]
187 }
188 k[0] = 0.0f;
189 at[0] = 0.0f;
190 for(i=1; i<=p; i++)
191 {
192 k[i] = -r[i];
193
194 for(j=1; j<i; j++)
195 {
196 at[j] = a[j];
197 k[i] -= a[j] * r[i-j];
198 }
199 if(fabs(e) < 1.0e-20f) { e = 0.0f; break; }
200 k[i] /= e;
201
202 a[i] = k[i];
203 for(j=1; j<i; j++) a[j] = at[j] + k[i] * at[i-j];
204
205 e *= 1.0f - k[i] * k[i];
206 }
207
208 if(e < 1.0e-20f) e = 0.0f;
209 *g = sqrtf(e);
210}

212Lfloat tTalkbox_tick(tTalkbox const v, Lfloat synth, Lfloat voice)
213{
214 int32_t p0=v->pos, p1 = (v->pos + v->N/2) % v->N;
215 Lfloat e=v->emphasis, w, o, x, fx=v->FX;
216 Lfloat p, q, h0=0.3f, h1=0.77f;
217
218 o = voice;
219 x = synth;
220
221
222
223 p = v->d0 + h0 * x; v->d0 = v->d1; v->d1 = x - h0 * p;
224 q = v->d2 + h1 * v->d4; v->d2 = v->d3; v->d3 = v->d4 - h1 * q;
225 v->d4 = x;
226 x = p + q;
227
228 if(v->K++)
229 {
230 v->K = 0;
231
232 v->car0[p0] = v->car1[p1] = x; //carrier input
233
234 x = o - e; e = o; //6dB/oct pre-emphasis
235
236 w = v->window[p0]; fx = v->buf0[p0] * w; v->buf0[p0] = x * w; //50% overlapping hanning windows
237 if(++p0 >= v->N) { tTalkbox_lpc(v->buf0, v->car0, v->dl, v->Rt, v->N, v->O, v->warpFactor, v->warpOn, v->k, v->freeze, &v->G); p0 = 0; }
238
239 w = 1.0f - w; fx += v->buf1[p1] * w; v->buf1[p1] = x * w;
240 if(++p1 >= v->N) { tTalkbox_lpc(v->buf1, v->car1, v->dl, v->Rt, v->N, v->O, v->warpFactor, v->warpOn, v->k, v->freeze, &v->G); p1 = 0; }
241 }
242
243 p = v->u0 + h0 * fx; v->u0 = v->u1; v->u1 = fx - h0 * p;
244 q = v->u2 + h1 * v->u4; v->u2 = v->u3; v->u3 = v->u4 - h1 * q;
245 v->u4 = fx;
246 x = p + q;
247
248 o = x;
249
250 v->emphasis = e;
251 v->pos = p0;
252 v->FX = fx;
253
254
255 return o;
256}

259void tTalkbox_lpc(Lfloat *buf, Lfloat *car, double* dl, double* Rt, int32_t n, int32_t o, Lfloat warp, int warpOn, Lfloat *k, int freeze, Lfloat *G)
260{
261 Lfloat z[ORD_MAX], r[ORD_MAX], x;
262 int32_t i, j, nn=n;
263
264 if (warpOn == 0)
265 {
266 for(j=0; j<=o; j++, nn--) //buf[] is already emphasized and windowed
267 {
268 z[j] = r[j] = 0.0f;
269 for(i=0; i<nn; i++) r[j] += buf[i] * buf[i+j]; //autocorrelation
270 }
271 }
272 else
273 {
274 for(j=0; j<=o; j++, nn--) //buf[] is already emphasized and windowed
275 {
276 z[j] = r[j] = 0.0f;
277 }
278 tTalkbox_warpedAutocorrelate(buf, dl, Rt, n, r, o, warp);
279 }
280
281 r[0] *= 1.001f; //stability fix
282
283 Lfloat min = 0.000001f;
284 if (!freeze)
285 {
286 if(r[0] < min)
287 {
288 for(i=0; i<n; i++)
289 {
290 buf[i] = 0.0f;
291 }
292 return;
293 }
294
295 tTalkbox_lpcDurbin(r, o, k, G); //calc reflection coeffs
296
297 //this is for stability to keep reflection coefficients inside the unit circle
298 //in mda's code it's .995 but in Harma's papers I've seen 0.998. just needs to be less than 1 it seems but maybe some wiggle room to avoid instability from Lfloating point precision -JS
299 for(i=0; i<=o; i++)
300 {
301 if(k[i] > 0.998f) k[i] = 0.998f; else if(k[i] < -0.998f) k[i] = -.998f;
302 }
303 }
304 for(i=0; i<n; i++)
305 {
306 x = G[0] * car[i];
307 for(j=o; j>0; j--) //lattice filter
308 {
309 x -= k[j] * z[j-1];
310 z[j] = z[j-1] + k[j] * x;
311 }
312 buf[i] = z[0] = x; //output buf[] will be windowed elsewhere
313 }
314}

627 {
628 if(k[i] > 0.998f) k[i] = 0.998f; else if(k[i] < -0.998f) k[i] = -.998f;
629 }
630 }

632 {
633 x = G[0] * car[i];
634 for(int j = o; j > 0; j--) //lattice filter
635 {
636 x -= k[j] * z[j-1];
637 z[j] = z[j-1] + k[j] * x;
638 }
639 buf[i] = z[0] = x; //output buf[] will be windowed elsewhere
640 }

716

793Lfloat tVocoder_tick (tVocoder const v, Lfloat synth, Lfloat voice)
794{
795 Lfloat a, b, o=0.0f, aa, bb, oo = v->kout, g = v->gain, ht = v->thru, hh = v->high, tmp;
796 uint32_t i, k = v->kval, nb = v->nbnd;
797
798 a = voice; //speech
799 b = synth; //synth
800
801 tmp = a - v->f[0][7]; //integrate modulator for HF band and filter bank pre-emphasis
802 v->f[0][7] = a;
803 a = tmp;
804
805 if(tmp<0.0f) tmp = -tmp;
806 v->f[0][11] -= v->f[0][12] * (v->f[0][11] - tmp); //high band envelope
807 o = v->f[0][11] * (ht * a + hh * (b - v->f[0][3])); //high band + high thru
808
809 v->f[0][3] = b; //integrate carrier for HF band
810
811 if(++k & 0x1) //this block runs at half sample rate
812 {
813 oo = 0.0f;
814 aa = a + v->f[0][9] - v->f[0][8] - v->f[0][8]; //apply zeros here instead of in each reson
815 v->f[0][9] = v->f[0][8]; v->f[0][8] = a;
816 bb = b + v->f[0][5] - v->f[0][4] - v->f[0][4];
817 v->f[0][5] = v->f[0][4]; v->f[0][4] = b;
818
819 for(i=1; i<nb; i++) //filter bank: 4th-order band pass
820 {
821 tmp = v->f[i][0] * v->f[i][3] + v->f[i][1] * v->f[i][4] + bb;
822 v->f[i][4] = v->f[i][3];
823 v->f[i][3] = tmp;
824 tmp += v->f[i][2] * v->f[i][5] + v->f[i][1] * v->f[i][6];
825 v->f[i][6] = v->f[i][5];
826 v->f[i][5] = tmp;
827
828 tmp = v->f[i][0] * v->f[i][7] + v->f[i][1] * v->f[i][8] + aa;
829 v->f[i][8] = v->f[i][7];
830 v->f[i][7] = tmp;
831 tmp += v->f[i][2] * v->f[i][9] + v->f[i][1] * v->f[i][10];
832 v->f[i][10] = v->f[i][9];
833 v->f[i][9] = tmp;
834
835 if(tmp<0.0f) tmp = -tmp;
836 v->f[i][11] -= v->f[i][12] * (v->f[i][11] - tmp);
837 oo += v->f[i][5] * v->f[i][11];
838 }
839 }
840 o += oo * g; //effect of interpolating back up to Fs would be minimal (aliasing >16kHz)
841
842 v->kout = oo;
843 v->kval = k & 0x1;
844#ifdef NO_DENORMAL_CHECK
845#else
846 if(fabs(v->f[0][11])<1.0e-10) v->f[0][11] = 0.0f; //catch HF envelope denormal
847
848 for(i=1;i<nb;i++)
849 if(fabs(v->f[i][3])<1.0e-10 || fabs(v->f[i][7])<1.0e-10)
850 for(k=3; k<12; k++) v->f[i][k] = 0.0f; //catch reson & envelope denormals
851#endif
852 if(fabs(o)>10.0f) tVocoder_suspend(v); //catch instability
853
854 return o;
855}

860
861 for(i=0; i<v->nbnd; i++) for(j=3; j<12; j++) v->f[i][j] = 0.0f; //zero band filters and envelopes
862 v->kout = 0.0f;
863 v->kval = 0;

1125/*
1126 Function pitchdown() is called to read samples from the delay buffer when pitch
1127 factor is between 0.25 and 1. The read pointer lags behind because of the slowed
1128 down speed, and it must jump forward towards the write pointer soon as there is
1129 sufficient space to jump. That is, if there is at least one period of the input
1130 signal between read pointer and write pointer. When short periods follow up on
1131 long periods, the read pointer may have space to jump over more than one period
1132 lenghts. Jump length must be [periodlength ^ 2] in any case.
1133
1134 A linear crossfade function joins the jump-from point with the jump-to point.
1135 The crossfade must be completed before another read pointer jump is allowed.
1136 Length of the crossfade function is stored as a number of samples in terms of
1137 the input sample rate. This length is dynamically translated
1138 to a crossfade length expressed in output reading rate, according to pitch
1139 factor which can change before the crossfade is completed. Crossfade length does
1140 not cover an invariable length in periods for all pitch transposition factors.
1141 For pitch factors from 0.5 till 1, crossfade length is stretched in the
1142 output just as much as the signal itself, as crossfade speed is set to equal
1143 pitch factor. For pitch factors below 0.5, the read pointer wants to jump
1144 forward before one period is read, therefore the crossfade length as expressed
1145 in output periods must be shorter. Crossfade speed is set to [1 - pitchfactor]
1146 for those cases. Pitch factor 0.5 is the natural switch point between crossfade
1147 speeds [pitchfactor] and [1 - pitchfactor] because 0.5 == 1 - 0.5. The crossfade
1148 speed modification for pitch factors below 0.5 also means that much of the
1149 original signal content will be skipped.
1150 */
1151
1152
1153static void pitchdown(_tSOLAD* const w, Lfloat *out)
1154{
1155 int n = w->blocksize;
1156 Lfloat refindex = (Lfloat)(w->timeindex + w->loopSize); // no negative values!
1157 Lfloat pitchfactor = w->pitchfactor;
1158 Lfloat period = w->period;
1159 Lfloat readlag = w->readlag;
1160 Lfloat readlagstep = 1 - pitchfactor;
1161 Lfloat jump = w->jump;
1162 Lfloat xfadevalue = w->xfadevalue;
1163 Lfloat xfadelength = w->xfadelength;
1164 Lfloat xfadespeed, xfadestep, readindex, outputsample;
1165
1166 if(pitchfactor > 0.5) xfadespeed = pitchfactor;
1167 else xfadespeed = 1 - pitchfactor;
1168 xfadestep = xfadespeed / xfadelength;
1169
1170 while(n--)
1171 {
1172 if(readlag > period) // check if read pointer may jump forward...
1173 {
1174 if(xfadevalue <= 0) // ...but do not interrupt crossfade
1175 {
1176 jump = period; // jump forward
1177 while((jump * 2) < readlag) jump *= 2; // use available space
1178 readlag -= jump; // reduce read pointer lag
1179 xfadevalue = 1; // start crossfade
1180 xfadelength = period - 1;
1181 xfadestep = xfadespeed / xfadelength;
1182 }
1183 }
1184
1185 readindex = refindex - readlag;
1186 outputsample = read_sample(w, readindex);
1187
1188 if(xfadevalue > 0)
1189 {
1190 outputsample *= (1 - xfadevalue); // fadein
1191 outputsample += read_sample(w, readindex - jump) * xfadevalue; // fadeout
1192 xfadevalue -= xfadestep;
1193 }
1194
1195 *out++ += outputsample;
1196 refindex += 1;
1197 readlag += readlagstep;
1198 }
1199
1200 w->jump = jump; // state variables
1201 w->readlag = readlag;
1202 w->xfadevalue = xfadevalue;
1203 w->xfadelength = xfadelength;
1204}
1205
1206
1207/*
1208 Function pitchup() for pitch factors above 1 is more complicated than
1209 pitchdown(). The read pointer increments faster than the write pointer and a
1210 backward jump must happen in time, reckoning with the crossfade region. The read
1211 pointer backward jump length is always one period. In order to minimize the area
1212 of signal duplicates, crossfade length is aimed at [period / pitchfactor].
1213 This leads to a crossfade speed of [pitchfactor * pitchfactor].
1214
1215 Some samples for the fade out (but not all of them) must already be in the
1216 buffer, otherwise we will run out of input samples before the crossfade is
1217 completed. The ratio of past samples and future samples for a crossfade of any
1218 length is as follows:
1219
1220 past samples: xfadelength * (1 - 1 / pitchfactor)
1221 future samples: xfadelength * (1 / pitchfactor)
1222
1223 For example in the case of pitch factor 1.5 this would be:
1224
1225 past samples: xfadelength * (1 - 1 / 1.5) = xfadelength * 1 / 3
1226 future samples: xfadelength * (1 / 1.5) = xfadelength * 2 / 3
1227
1228 In the case of pitch factor 4 this would be:
1229
1230 past samples: xfadelength * (1 - 1 / 4) = xfadelength * 3 / 4
1231 future samples: xfadelength * (1 / 4) = xfadelength * 1 / 4
1232
1233 The read pointer lag must therefore preserve a minimum dependent on pitch
1234 factor. The minimum is called 'limit' here:
1235
1236 limit = period * (pitchfactor - 1) / pitchfactor * pitchfactor
1237
1238 Components of this expression are combined to reuse them in operations, while
1239 (pitchfactor - 1) is changed to (pitchfactor - 0.99) to avoid numerical
1240 resolution issues for pitch factors slightly above 1:
1241
1242 xfadespeed = pitchfactor * pitchfactor
1243 limitfactor = (pitchfactor - 0.99) / xfadespeed
1244 limit = period * limitfactor
1245
1246 When read lag is smaller than this limit, the read pointer must preferably
1247 jump backward, unless a previous crossfade is not yet completed. Crossfades must
1248 preferably be completed, unless the read pointer lag becomes smaller than zero.
1249 With fluctuating period lengths and pitch factors, the readpointer lag limit may
1250 change from one input block to the next in such a way that the actual lag is
1251 suddenly much smaller than the limit, and the intended crossfade length can not
1252 be applied. Therefore the crossfade length is simply calculated from the
1253 available amount of samples for all cases, like so:
1254
1255 xfadelength = readlag / limitfactor
1256
1257 For most occurrences, this will amount to a crossfade length reduced to
1258 [period / pitchfactor] in the output for pitch factors above 1, while in some
1259 cases it will be considerably shorter. Fortunately, an incidental aberration of
1260 the intended crossfade length hardly ever creates an audible artifact. The
1261 reason to specify preferred crossfade length according to pitch factor is to
1262 minimize the impression of echoes without sacrificing too much of the signal
1263 content. The readpointer jump length remains one period in any case.
1264
1265 Sometimes, the input signal periodicity may decrease substantially between one
1266 signal block and the next. In such cases it may be possible for the read pointer
1267 to jump forward and reduce latency. For every signal block, a check on this
1268 possibility is done. A previous crossfade must be completed before a forward
1269 jump is allowed.
1270 */
1271static void pitchup(_tSOLAD* const w, Lfloat *out)
1272{
1273 int n = w->blocksize;
1274 Lfloat refindex = (Lfloat)(w->timeindex + w->loopSize); // no negative values
1275 Lfloat pitchfactor = w->pitchfactor;
1276 Lfloat period = w->period;
1277 Lfloat readlag = w->readlag;
1278 Lfloat jump = w->jump;
1279 Lfloat xfadevalue = w->xfadevalue;
1280 Lfloat xfadelength = w->xfadelength;
1281
1282 Lfloat readlagstep = pitchfactor - 1;
1283 Lfloat xfadespeed = pitchfactor * pitchfactor;
1284 Lfloat xfadestep = xfadespeed / xfadelength;
1285 Lfloat limitfactor = (pitchfactor - (Lfloat)0.99) / xfadespeed;
1286 Lfloat limit = period * limitfactor;
1287 Lfloat readindex, outputsample;
1288
1289 if((readlag > (period + 2 * limit)) & (xfadevalue < 0))
1290 {
1291 jump = period; // jump forward
1292 while((jump * 2) < (readlag - 2 * limit)) jump *= 2; // use available space
1293 readlag -= jump; // reduce read pointer lag
1294 xfadevalue = 1; // start crossfade
1295 xfadelength = period - 1;
1296 xfadestep = xfadespeed / xfadelength;
1297 }
1298
1299 while(n--)
1300 {
1301 if(readlag < limit) // check if read pointer should jump backward...
1302 {
1303 if((xfadevalue < 0) | (readlag < 0)) // ...but try not to interrupt crossfade
1304 {
1305 xfadelength = readlag / limitfactor;
1306 if(xfadelength < 1) xfadelength = 1;
1307 xfadestep = xfadespeed / xfadelength;
1308
1309 jump = -period; // jump backward
1310 readlag += period; // increase read pointer lag
1311 xfadevalue = 1; // start crossfade
1312 }
1313 }
1314
1315 readindex = refindex - readlag;
1316 outputsample = read_sample(w, readindex);
1317
1318 if(xfadevalue > 0)
1319 {
1320 outputsample *= (1 - xfadevalue);
1321 outputsample += read_sample(w, readindex - jump) * xfadevalue;
1322 xfadevalue -= xfadestep;
1323 }
1324
1325 *out++ += outputsample;
1326 refindex += 1;
1327 readlag -= readlagstep;
1328 }
1329
1330 w->readlag = readlag; // state variables
1331 w->jump = jump;
1332 w->xfadelength = xfadelength;
1333 w->xfadevalue = xfadevalue;
1334}
1335
1336// read one sample from delay buffer, with linear interpolation
1337static inline Lfloat read_sample(_tSOLAD* const w, Lfloat Lfloatindex)
1338{
1339 int index = (int)Lfloatindex;
1340 Lfloat fraction = Lfloatindex - (Lfloat)index;
1341 Lfloat *buf = w->delaybuf;
1342 index &= (w->loopSize - 1);
1343
1344 return (buf[index] + (fraction * (buf[index+1] - buf[index])));
1345}

1761{

1763 in = tHighpass_tick(fs->hp, in * fs->intensity);
1764
1765
1766 Lfloat fa, fb, fc, foma, falph, ford, flamb, tf, fk;
1767

1771 flamb = fs->flamb;
1772
1773 tf = in;
1774

1777 fb = fa;
1778 for(int i = 0; i < ford; i++)
1779 {
1780 fs->fsig[i] = fa*fa*foma + fs->fsig[i]*falph;
1781 fc = (fb - fs->fc[i])*flamb + fs->fb[i];
1782 fs->fc[i] = fc;

1785 fs->fk[i] = fk;

1787 tf = tf*foma + fs->fsmooth[i]*falph;
1788 fs->fsmooth[i] = tf;
1789 fs->fbuff[i] = tf;
1790 fb = fc - tf*fa;
1791 fa = fa - tf*fc;
1792 }
