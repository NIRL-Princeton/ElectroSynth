117void tEnvelope_on (tEnvelope const env, Lfloat velocity)
118{
119 if (env->inAttack || env->inDecay) // In case envelope retriggered while it is still happening.
120 {
121 env->rampPhase = 0;
122 env->inRamp = 1;
123 env->rampPeak = env->next;
124 } else // Normal start.
125 {
126 env->inAttack = 1;
127 }
128
129
130 env->attackPhase = 0;
131 env->decayPhase = 0;
132 env->inDecay = 0;
133 env->gain = velocity;

136Lfloat tEnvelope_tick (tEnvelope const env)
137{
138 if (env->inRamp) {
139 if (env->rampPhase > UINT16_MAX) {
140 env->inRamp = 0;
141 env->inAttack = 1;
142 env->next = 0.0f;
143 } else {
144 env->next = env->rampPeak * env->exp_buff[(uint32_t) env->rampPhase];
145 }
146
147 env->rampPhase += env->rampInc;
148 }
149
150 if (env->inAttack) {
151
152 // If attack done, time to turn around.
153 if (env->attackPhase > UINT16_MAX) {
154 env->inDecay = 1;
155 env->inAttack = 0;
156 env->next = env->gain * 1.0f;
157 } else {
158 // do interpolation !
159 env->next = env->gain * env->exp_buff[UINT16_MAX -
160 (uint32_t) env->attackPhase]; // inverted and backwards to get proper rising exponential shape/perception
161 }
162
163 // Increment envelope attack.
164 env->attackPhase += env->attackInc;
165
166 }
167
168 if (env->inDecay) {
169
170 // If decay done, finish.
171 if (env->decayPhase >= UINT16_MAX) {
172 env->inDecay = 0;
173
174 if (env->loop) {
175 env->attackPhase = 0;
176 env->decayPhase = 0;
177 env->inAttack = 1;
178 } else {
179 env->next = 0.0f;
180 }
181
182 } else {
183
184 env->next = env->gain * (env->exp_buff[(uint32_t) env->decayPhase]); // do interpolation !
185 }
186
187 // Increment envelope decay;
188 env->decayPhase += env->decayInc;
189 }
190
191 return env->next;
192}

384 adsr->next = 0.0f;

387 }
388
389 adsr->rampPhase += adsr->rampInc;
390 }
391
392 if (adsr->inAttack) {
393
394 // If attack done, time to turn around.
395 if (adsr->attackPhase > UINT16_MAX) {
396 adsr->inDecay = 1;
397 adsr->inAttack = 0;
398 adsr->next = adsr->gain * 1.0f;
399 } else {
400 // do interpolation !
401 adsr->next = adsr->gain * adsr->exp_buff[UINT16_MAX -
402 (uint32_t) adsr->attackPhase]; // inverted and backwards to get proper rising exponential shape/perception
403 }
404
405 // Increment ADSR attack.
406 adsr->attackPhase += adsr->attackInc;
407
408 }
409
410 if (adsr->inDecay) {
411
412 // If decay done, sustain.
413 if (adsr->decayPhase >= UINT16_MAX) {
414 adsr->inDecay = 0;
415 adsr->inSustain = 1;
416 adsr->next = adsr->gain * adsr->sustain;
417 } else {
418 adsr->next = (adsr->gain *
419 (adsr->sustain + ((adsr->exp_buff[(uint32_t) adsr->decayPhase]) * (1.0f - adsr->sustain)))) *
420 adsr->leakFactor; // do interpolation !
421 }
422
423 // Increment ADSR decay.
424 adsr->decayPhase += adsr->decayInc;
425 }
426
427 if (adsr->inSustain) {
428 adsr->next = adsr->next * adsr->leakFactor;
429 }
430
431 if (adsr->inRelease) {
432 // If release done, finish.
433 if (adsr->releasePhase >= UINT16_MAX) {
434 adsr->inRelease = 0;
435 adsr->next = 0.0f;
436 } else {
437
438 adsr->next = adsr->releasePeak * (adsr->exp_buff[(uint32_t) adsr->releasePhase]); // do interpolation !
439 }
440
441 // Increment envelope release;
442 adsr->releasePhase += adsr->releaseInc;
443 }
444
445
446 return adsr->next;
447}

466Lfloat calcADSR3Coef (Lfloat rate, Lfloat targetRatio)
467{
468 return (rate <= 0.0f) ? 0.0f : expf(-logf((1.0f + targetRatio) / targetRatio) / rate);
469}

477void tADSRS_initToPool (tADSRS *const adsrenv, Lfloat attack, Lfloat decay,
478 Lfloat sustain, Lfloat release,
479 tMempool *const mp)
480{
481 _tMempool *m = *mp;
482 _tADSRS *adsr = *adsrenv = (_tADSRS *) mpool_alloc(sizeof(_tADSRS), m);
483 adsr->mempool = m;
484
485 LEAF *leaf = adsr->mempool->leaf;
486
487 adsr->sampleRate = leaf->sampleRate;
488 adsr->sampleRateInMs = adsr->sampleRate * 0.001f;
489 adsr->targetRatioA = 0.3f;
490 adsr->targetRatioDR = 0.0001f;
491 adsr->attack = attack;
492 adsr->attackRate = attack * adsr->sampleRateInMs;
493 adsr->attackCoef = calcADSR3Coef(attack * adsr->sampleRateInMs, adsr->targetRatioA);
494 adsr->attackBase = (1.0f + adsr->targetRatioA) * (1.0f - adsr->attackCoef);
495
496 adsr->decay = decay;
497 adsr->decayRate = decay * adsr->sampleRateInMs;
498 adsr->decayCoef = calcADSR3Coef(decay * adsr->sampleRateInMs, adsr->targetRatioDR);
499 adsr->decayBase = (adsr->sustainLevel - adsr->targetRatioDR) * (1.0f - adsr->decayCoef);
500
501 adsr->sustainLevel = sustain;
502 adsr->decayBase = (adsr->sustainLevel - adsr->targetRatioDR) * (1.0f - adsr->decayCoef);
503
504 adsr->release = release;
505 adsr->releaseRate = release * adsr->sampleRateInMs;
506 adsr->releaseCoef = calcADSR3Coef(release * adsr->sampleRateInMs, adsr->targetRatioDR);
507 adsr->releaseBase = -adsr->targetRatioDR * (1.0f - adsr->releaseCoef);
508
509 adsr->state = env_idle;
510 adsr->gain = 1.0f;
511 adsr->targetGainSquared = 1.0f;
512 adsr->factor = 0.01f;
513 adsr->oneMinusFactor = 0.99f;
514 adsr->output = 0.0f;
515
516 adsr->baseLeakFactor = 1.0f;
517 adsr->leakFactor = 1.0f;
518
519 adsr->invSampleRate = leaf->invSampleRate;
520}

577Lfloat tADSRS_tick (tADSRS const adsr)
578{
579 switch (adsr->state) {
580 case env_idle:
581 break;
582 case env_attack:
583 adsr->output = adsr->attackBase + adsr->output * adsr->attackCoef;
584 if (adsr->output >= 1.0f) {
585 adsr->output = 1.0f;
586 adsr->state = env_decay;
587 }
588 break;
589 case env_decay:
590 adsr->output = adsr->decayBase + adsr->output * adsr->decayCoef * adsr->leakFactor;
591 if (adsr->output <= adsr->sustainLevel) {
592 adsr->output = adsr->sustainLevel;
593 adsr->state = env_sustain;
594 }
595 break;
596 case env_sustain:
597 adsr->output = adsr->output * adsr->leakFactor;
598 break;
599 case env_release:
600 adsr->output = adsr->releaseBase + adsr->output * adsr->releaseCoef;
601 if (adsr->output <= 0.0f) {
602 adsr->output = 0.0f;
603 adsr->state = env_idle;
604 }
605 default:
606 break;
607 }
608 //smooth the gain value -- this is not ideal, a retrigger while the envelope is still going with a new gain will cause a jump, although it will be smoothed quickly. Maybe doing the math so the range is computed based on the gain rather than 0.->1. is preferable? But that's harder to get the exponential curve right without a lookup.
609 adsr->gain = (adsr->factor * adsr->targetGainSquared) + (adsr->oneMinusFactor * adsr->gain);
610 return adsr->output * adsr->gain;
611}

636//initialize with an exponential function that decays -- i.e. a call to LEAF_generate_exp(expBuffer, 0.001f, 0.0f, 1.0f, -0.0008f, EXP_BUFFER_SIZE);
637//times are in ms
638void
639tADSRT_initToPool(tADSRT *const adsrenv, Lfloat attack, Lfloat decay, Lfloat sustain,
640 Lfloat release, Lfloat *expBuffer, int bufferSize, tMempool *const mp)
641{
642 _tMempool *m = *mp;
643 _tADSRT *adsr = *adsrenv = (_tADSRT *) mpool_alloc(sizeof(_tADSRT), m);
644 adsr->mempool = m;
645
646 LEAF *leaf = adsr->mempool->leaf;
647
648 adsr->exp_buff = expBuffer;
649 adsr->buff_size = bufferSize;
650 adsr->buff_sizeMinusOne = bufferSize - 1;
651
652 adsr->sampleRate = leaf->sampleRate;
653 adsr->bufferSizeDividedBySampleRateInMs = adsr->buff_size / (adsr->sampleRate * 0.001f);
654
655 if (attack < 0.0f)
656 attack = 0.0f;
657
658 if (decay < 0.0f)
659 decay = 0.0f;
660
661 if (sustain > 1.0f)
662 sustain = 1.0f;
663 if (sustain < 0.0f)
664 sustain = 0.0f;
665
666 if (release < 0.0f)
667 release = 0.0f;
668
669 adsr->next = 0.0f;
670
671 adsr->whichStage = env_idle;
672
673 adsr->sustain = sustain;
674
675 adsr->attack = attack;
676 adsr->decay = decay;
677 adsr->release = release;

810#ifdef ITCMRAM
811Lfloat __attribute__ ((section(".itcmram"))) __attribute__ ((aligned (32))) tADSRT_tick(tADSRT* const adsrenv)
812#else
813
814Lfloat tADSRT_tick (tADSRT const adsr)
815#endif
816{
817 switch (adsr->whichStage) {
818 case env_ramp:
819 if (adsr->rampPhase > adsr->buff_sizeMinusOne) {
820 adsr->whichStage = env_attack;
821 adsr->next = 0.0f;
822 } else {
823 uint32_t intPart = (uint32_t) adsr->rampPhase;
824 Lfloat LfloatPart = adsr->rampPhase - intPart;
825 Lfloat secondValue;
826 if (adsr->rampPhase + 1.0f > adsr->buff_sizeMinusOne) {
827 secondValue = 0.0f;
828 } else {
829 secondValue = adsr->exp_buff[(uint32_t) ((adsr->rampPhase) + 1)];
830 }
831 adsr->next =
832 adsr->rampPeak * LEAF_interpolation_linear(adsr->exp_buff[intPart], secondValue, LfloatPart);
833 }
834
835 adsr->rampPhase += adsr->rampInc;
836 break;
837
838
839 case env_attack:
840
841 // If attack done, time to turn around.
842 if (adsr->attackPhase > adsr->buff_sizeMinusOne) {
843 adsr->whichStage = env_decay;
844 adsr->next = adsr->gain;
845 } else {
846 // do interpolation !
847 uint32_t intPart = (uint32_t) adsr->attackPhase;
848 Lfloat LfloatPart = adsr->attackPhase - intPart;
849 Lfloat secondValue;
850 if (adsr->attackPhase + 1.0f > adsr->buff_sizeMinusOne) {
851 secondValue = 0.0f;
852 } else {
853 secondValue = adsr->exp_buff[(uint32_t) ((adsr->attackPhase) + 1)];
854 }
855
856 adsr->next = adsr->gain * (1.0f - LEAF_interpolation_linear(adsr->exp_buff[intPart], secondValue,
857 LfloatPart)); // inverted and backwards to get proper rising exponential shape/perception
858 }
859
860 // Increment ADSR attack.
861 adsr->attackPhase += adsr->attackInc;
862 break;
863
864 case env_decay:
865
866 // If decay done, sustain.
867 if (adsr->decayPhase > adsr->buff_sizeMinusOne) {
868 adsr->whichStage = env_sustain;
869 adsr->next = adsr->gain * adsr->sustain;
870 adsr->sustainWithLeak = 1.0f;
871 } else {
872 uint32_t intPart = (uint32_t) adsr->decayPhase;
873 Lfloat LfloatPart = adsr->decayPhase - intPart;
874 Lfloat secondValue;
875 if (adsr->decayPhase + 1.0f > adsr->buff_sizeMinusOne) {
876 secondValue = 0.0f;
877 } else {
878 secondValue = adsr->exp_buff[(uint32_t) ((adsr->decayPhase) + 1)];
879 }
880 Lfloat interpValue = (LEAF_interpolation_linear(adsr->exp_buff[intPart], secondValue, LfloatPart));
881 adsr->next = (adsr->gain * (adsr->sustain + (interpValue * (1.0f - adsr->sustain)))) *
882 adsr->leakFactor; // do interpolation !
883 }
884
885 // Increment ADSR decay.
886 adsr->decayPhase += adsr->decayInc;
887 break;
888
889 case env_sustain:
890 adsr->next = adsr->sustain * adsr->gain * (adsr->leakFactor * adsr->sustainWithLeak);
891 break;
892
893 case env_release:
894 // If release done, finish.
895 if (adsr->releasePhase > adsr->buff_sizeMinusOne) {
896 adsr->whichStage = env_idle;
897 adsr->next = 0.0f;
898 } else {
899 uint32_t intPart = (uint32_t) adsr->releasePhase;
900 Lfloat LfloatPart = adsr->releasePhase - intPart;
901 Lfloat secondValue;
902 if (adsr->releasePhase + 1.0f > adsr->buff_sizeMinusOne) {
903 secondValue = 0.0f;
904 } else {
905 secondValue = adsr->exp_buff[(uint32_t) ((adsr->releasePhase) + 1)];
906 }
907 adsr->next = adsr->releasePeak * (LEAF_interpolation_linear(adsr->exp_buff[intPart], secondValue,
908 LfloatPart)); // do interpolation !
909 }
910
911 // Increment envelope release;
912 adsr->releasePhase += adsr->releaseInc;
913 break;
914 }
915 return adsr->next;
916}

920#else

923#endif
924{

928 adsr->whichStage = env_attack;
929 adsr->next = 0.0f;

936
937

987 }
