73 inline void positionBar(int index, float x, float y, float width, float height) {
74 bar_data_[kFloatsPerBar * index] = x;
75 bar_data_[kFloatsPerBar * index + 1] = y;
76
77 bar_data_[kFloatsPerBar * index + kFloatsPerVertex] = x + width;
78 bar_data_[kFloatsPerBar * index + kFloatsPerVertex + 1] = y;
79
80 bar_data_[kFloatsPerBar * index + 2 * kFloatsPerVertex] = x;
81 bar_data_[kFloatsPerBar * index + 2 * kFloatsPerVertex + 1] = y + height;
82
83 bar_data_[kFloatsPerBar * index + 3 * kFloatsPerVertex] = x + width;
84 bar_data_[kFloatsPerBar * index + 3 * kFloatsPerVertex + 1] = y + height;
85 dirty_ = true;

93 force_inline float scaledYAt(int index) {
94 float value = yAt(index) * 0.5f + 0.5f;
95 if (square_scale_)
96 value *= value;
97 if (power_scale_)
98 value /= std::max(index, 1) / kScaleConstant;

103 force_inline void setScaledY(int index, float val) {
104 float value = val;
105
106 if (power_scale_)
107 value *= std::max(index, 1) / kScaleConstant;
108 if (square_scale_)
109 value = sqrtf(value);
110
111 setY(index, 2.0f * value - 1.0f);
112 }
