35 template <typename T, typename Pair, typename... Rest>
36 static constexpr int get_impl() {
37 if constexpr (std::is_same_v<T, typename Pair::type>) {
38 return Pair::value;
39 } else {
40 return get_impl<T, Rest...>();
41 }
42 }

46 static constexpr int get_impl() {
47 static_assert(sizeof...(Pairs) > 0, "Type not found in map");
