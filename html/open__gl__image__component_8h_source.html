125 image_component_->setComponent(this);
126 addListener(this);
127 }
128
129 bool keyPressed(const juce::KeyPress& key) override {
130 bool result = juce::TextEditor::keyPressed(key);
131 redoImage();
132 return result;
133 }
134
135 void textEditorTextChanged(juce::TextEditor&) override { redoImage(); }
136 void textEditorFocusLost(juce::TextEditor&) override { redoImage(); }
137
138 virtual void mouseDrag(const juce::MouseEvent& e) override {
139 juce::TextEditor::mouseDrag(e);
140 redoImage();
141 }
142
143 void applyFont() {
144 juce::Font font;
145 if (monospace_)
146 font = Fonts::instance()->monospace().withPointHeight(getHeight() / 2.0f);
147 else
148 font = Fonts::instance()->proportional_light().withPointHeight(getHeight() / 2.0f);
149
150 applyFontToAllText(font);
151 redoImage();
152 }
153
154 void visibilityChanged() override {
155 juce::TextEditor::visibilityChanged();
156
157 if (isVisible() && !isMultiLine())
158 applyFont();
159 }
160
161 void resized() override {
162 juce::TextEditor::resized();
163 if (isMultiLine()) {
164 float indent = image_component_->findValue(Skin::kLabelBackgroundRounding);
165 setIndents(indent, indent);
166 return;
167 }
168
169 if (monospace_)
170 setIndents(getHeight() * 0.2, getHeight() * 0.17);
171 else
172 setIndents(getHeight() * 0.2, getHeight() * 0.15);
173 if (isVisible())
174 applyFont();
175 }
176
177 void setMonospace() {
178 monospace_ = true;
179 }
180
181 private:
182 bool monospace_;
183
184 JUCE_DECLARE_NON_COPYABLE_WITH_LEAK_DETECTOR(OpenGlTextEditor)

187class PlainTextComponent : public OpenGlImageComponent {
188 public:
189 enum FontType {
190 kTitle,
191 kLight,
192 kRegular,
193 kMono,
194 kNumFontTypes
195 };
196
197 PlainTextComponent(juce::String name, juce::String text) : OpenGlImageComponent(name), text_(std::move(text)),
198 text_size_(1.0f), font_type_(kRegular),
199 justification_(juce::Justification::centred),
200 buffer_(0) {
201 setInterceptsMouseClicks(false, false);
202 }
203
204 void resized() override {
205 OpenGlImageComponent::resized();
206 redrawImage(true);
207 }
208
209 void setText(juce::String text) {
210 if (text_ == text)
211 return;
212
213 text_ = text;
214 redrawImage(true);
215 }
216
217 juce::String getText() const { return text_; }
218
219 void paintToImage(juce::Graphics& g) override {
220 g.setColour(juce::Colours::white);
221
222 if (font_type_ == kTitle)
223 g.setFont(Fonts::instance()->proportional_title().withPointHeight(text_size_));
224 else if (font_type_ == kLight)
225 g.setFont(Fonts::instance()->proportional_light().withPointHeight(text_size_));
226 else if (font_type_ == kRegular)
227 g.setFont(Fonts::instance()->proportional_regular().withPointHeight(text_size_));
228 else
229 g.setFont(Fonts::instance()->monospace().withPointHeight(text_size_));
230
231 juce::Component* component = component_ ? component_ : this;
232
233 g.drawFittedText(text_, buffer_, 0, component->getWidth() - 2 * buffer_,
234 component->getHeight(), justification_, false);
235 }
236
237 void setTextSize(float size) {
238 text_size_ = size;
239 redrawImage(true);
240 }
241
242 void setFontType(FontType font_type) {
243 font_type_ = font_type;
244 }
245
246 void setJustification(juce::Justification justification) {
247 justification_ = justification;
248 }
249
250 void setBuffer(int buffer) { buffer_ = buffer; }
251
252 private:
253 juce::String text_;
254 float text_size_;
255 FontType font_type_;
256 juce::Justification justification_;
257 int buffer_;
258
259 JUCE_DECLARE_NON_COPYABLE_WITH_LEAK_DETECTOR(PlainTextComponent)
260};
