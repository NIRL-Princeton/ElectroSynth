81Lfloat t808Cowbell_tick(t808Cowbell const cowbell)
82{
83 Lfloat sample = 0.0f;
84
85 // Mix oscillators.
86 sample = (cowbell->oscMix * tSquare_tick(cowbell->p[0])) + ((1.0f-cowbell->oscMix) * tSquare_tick(cowbell->p[1]));
87
88 // Filter dive and filter.
89 tSVF_setFreq(cowbell->bandpassOsc, cowbell->filterCutoff + 1000.0f * tEnvelope_tick(cowbell->envFilter));
90 sample = tSVF_tick(cowbell->bandpassOsc,sample);
91
92 sample *= (0.9f * tEnvelope_tick(cowbell->envGain));
93
94 if (cowbell->useStick)
95 sample += (0.1f * tEnvelope_tick(cowbell->envStick) * tSVF_tick(cowbell->bandpassStick, tNoise_tick(cowbell->stick)));
96
97
98 sample = tHighpass_tick(cowbell->highpass, sample);
99
100 return sample;
101}

219Lfloat t808Hihat_tick(t808Hihat const hihat)
220{
221 Lfloat sample = 0.0f;
222 Lfloat gainScale = 0.1666f;
223
224 Lfloat myNoise = tNoise_tick(hihat->n);
225
226 tSquare_setFreq(hihat->p[0], ((2.0f + hihat->stretch) * hihat->freq));
227 tSquare_setFreq(hihat->p[1], ((3.00f + hihat->stretch) * hihat->freq));
228 tSquare_setFreq(hihat->p[2], ((4.16f + hihat->stretch) * hihat->freq));
229 tSquare_setFreq(hihat->p[3], ((5.43f + hihat->stretch) * hihat->freq));
230 tSquare_setFreq(hihat->p[4], ((6.79f + hihat->stretch) * hihat->freq));
231 tSquare_setFreq(hihat->p[5], ((8.21f + hihat->stretch) * hihat->freq));
232
233 for (int i = 0; i < 6; i++)
234 {
235 sample += tSquare_tick(hihat->p[i]);
236 }
237
238 sample *= gainScale;
239
240 sample = (hihat->oscNoiseMix * sample) + ((1.0f-hihat->oscNoiseMix) * myNoise);
241
242 sample = tSVF_tick(hihat->bandpassOsc, sample);
243
244 Lfloat myGain = tEnvelope_tick(hihat->envGain);
245 sample *= (myGain*myGain);//square the output gain envelope
246 sample = tHighpass_tick(hihat->highpass, sample);
247 sample += ((0.5f * tEnvelope_tick(hihat->envStick)) * tSVF_tick(hihat->bandpassStick, tNoise_tick(hihat->stick)));
248 sample = tanhf(sample * 2.0f);
249
250 return sample;
251}

428{
429 Lfloat tone[2];
430 for (int i = 0; i < 2; i++)
431 {
432 tTriangle_setFreq(snare->tone[i], snare->tone1Freq + (20.0f * tEnvelope_tick(snare->toneEnvOsc[i])));
433 tone[i] = tTriangle_tick(snare->tone[i]);
434
435 tSVF_setFreq(snare->toneLowpass[i], 2000.0f + (500.0f * tEnvelope_tick(snare->toneEnvFilter[i])));
436 tone[i] = tSVF_tick(snare->toneLowpass[i], tone[i]) * tEnvelope_tick(snare->toneEnvGain[i]);
437 }
438
439 Lfloat noise = tNoise_tick(snare->noiseOsc);
440 tSVF_setFreq(snare->noiseLowpass, snare->noiseFilterFreq + (1000.0f * tEnvelope_tick(snare->noiseEnvFilter)));
441 noise = tSVF_tick(snare->noiseLowpass, noise) * tEnvelope_tick(snare->noiseEnvGain);
442
443 Lfloat sample = (snare->toneNoiseMix)*(tone[0] * snare->toneGain[0] + tone[1] * snare->toneGain[1]) + (1.0f-snare->toneNoiseMix) * (noise * snare->noiseGain);
444 sample = tanhf(sample * 2.0f);
445 return sample;
446}
