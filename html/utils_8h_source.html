129 force_inline int ilog2(int value) {
130 #if defined(__GNUC__) || defined(__clang__)
131 constexpr int kMaxBitIndex = sizeof(int) * 8 - 1;
132 return kMaxBitIndex - __builtin_clz(std::max(value, 1));
133 #elif defined(_MSC_VER)
134 unsigned long result = 0;
135 _BitScanReverse(&result, value);
136 return result;
137 #else
138 int num = 0;
139 while (value >>= 1)
140 num++;
141 return num;
142 #endif
143 }
