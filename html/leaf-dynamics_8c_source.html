49void tCompressor_initToPool (tCompressor* const comp, tMempool* const mp)
50{
51 _tMempool* m = *mp;
52 _tCompressor* c = *comp = (_tCompressor*) mpool_alloc(sizeof(_tCompressor), m);
53 c->mempool = m;
54 LEAF* leaf = c->mempool->leaf;
55 c->sampleRate = leaf->sampleRate;
56
57 c->tauAttack = expf(-1.0f/(0.001f * 50.0f * c->sampleRate));
58 c->tauRelease = expf(-1.0f/(0.001f * 100.0f * c->sampleRate));
59
60 c->isActive = 0;
61 c->x_G[0] = 0.0f;
62 c->x_G[1] = 0.0f;
63 c->y_G[0] = 0.0f;

65 c->x_T[0] = 0.0f;
66 c->x_T[1] = 0.0f;

86Lfloat tCompressor_tick(tCompressor const c, Lfloat in)
87{
88 Lfloat slope, overshoot;
89
90 Lfloat in_db = LEAF_clip(-90.0f, fasteratodb(fastabsf(in)), 0.0f);
91 Lfloat out_db = 0.0f;
92
93 c->y_T[1] = c->y_T[0];
94
95 slope = 1.0f - c->invR; // feed-forward topology;
96
97 overshoot = in_db - c->T;
98
99
100 if (overshoot <= -(c->W))
101 {
102 out_db = in_db;
103 c->isActive = 0;
104 }
105 else if ((overshoot > -(c->W)) && (overshoot < (c->W)))
106 {
107 Lfloat squareit = (overshoot + c->W);
108 out_db = in_db + slope * ((squareit * squareit) * c->inv4W); // .^ 2 ???
109 c->isActive = 1;
110 }
111 else
112 {
113 out_db = in_db + slope * overshoot;
114 c->isActive = 1;
115 }
116/* // simpler hard-knee version
117 if (overshoot <= 0.0f)
118 {
119 out_db = in_db;
120 c->isActive = 0;
121 }
122 else
123 {
124 out_db = in_db + slope * overshoot;
125 c->isActive = 1;
126 }
127 */
128 c->x_T[0] = out_db - in_db;
129 if (c->x_T[0] > c->y_T[1])
130 c->y_T[0] = c->tauAttack * c->y_T[1] + (1.0f-c->tauAttack) * c->x_T[0];
131 else
132 c->y_T[0] = c->tauRelease * c->y_T[1] + (1.0f-c->tauRelease) * c->x_T[0];
133 Lfloat attenuation = fasterdbtoa(c->M - c->y_T[0]);
134 return attenuation * in;
135}

251{
