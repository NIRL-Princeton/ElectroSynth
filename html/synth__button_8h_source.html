134class OpenGlButtonComponent : public OpenGlComponent {
135 public:
136 static constexpr float kHoverInc = 0.2f;
137
138 enum ButtonStyle {
139 kTextButton,
140 kJustText,
141 kPowerButton,
142 kUiButton,
143 kLightenButton,
144 kNumButtonStyles
145 };
146
147 OpenGlButtonComponent(Button* button) : style_(kTextButton), button_(button),
148 show_on_colors_(true), primary_ui_button_(false),
149 down_(false), hover_(false), hover_amount_(0.0f),
150 background_(Shaders::kRoundedRectangleFragment), text_("text", "") {
151 background_.setTargetComponent(button);
152 background_.setColor(Colours::orange);
153 background_.setQuad(0, -1.0f, -1.0f, 2.0f, 2.0f);
154
155 addChildComponent(text_);
156 text_.setActive(false);
157 text_.setScissor(true);
158 text_.setComponent(button);
159 text_.setFontType(PlainTextComponent::kMono);
160 }
161
162 virtual void init(OpenGlWrapper& open_gl) override {
163 if (style_ == kPowerButton)
164 background_.setFragmentShader(Shaders::kCircleFragment);
165
166 background_.init(open_gl);
167 text_.init(open_gl);
168
169 setColors();
170 }
171
172 void setColors();
173
174 void renderTextButton(OpenGlWrapper& open_gl, bool animate);
175 void renderPowerButton(OpenGlWrapper& open_gl, bool animate);
176 void renderUiButton(OpenGlWrapper& open_gl, bool animate);
177 void renderLightenButton(OpenGlWrapper& open_gl, bool animate);
178 void incrementHover();
179
180 virtual void render(OpenGlWrapper& open_gl, bool animate) override {
181 if (style_ == kTextButton || style_ == kJustText)
182 renderTextButton(open_gl, animate);
183 else if (style_ == kPowerButton)
184 renderPowerButton(open_gl, animate);
185 else if (style_ == kUiButton)
186 renderUiButton(open_gl, animate);
187 else if (style_ == kLightenButton)
188 renderLightenButton(open_gl, animate);
189 }
190
191 void setText() {
192 String text = button_->getButtonText();
193 if (!text.isEmpty()) {
194 text_.setActive(true);
195 text_.setText(text);
196 }
197 }
198
199 void setDown(bool down) { down_ = down; }
200 void setHover(bool hover) { hover_ = hover; }
201
202 virtual void destroy(juce::OpenGLContext& open_gl) override {
203 background_.destroy(open_gl);
204 text_.destroy(open_gl);
205 }
206
207 void setJustification(Justification justification) { text_.setJustification(justification); }
208 void setStyle(ButtonStyle style) { style_ = style; }
209 void setShowOnColors(bool show) { show_on_colors_ = show; }
210 void setPrimaryUiButton(bool primary) { primary_ui_button_ = primary; }
211
212 void paintBackground(Graphics& g) override { }
213
214 OpenGlQuad& background() { return background_; }
215 PlainTextComponent& text() { return text_; }
216 ButtonStyle style() { return style_; }
217 void setScissorComponent(juce::Component *scissor_component) override
218 {
219 OpenGlComponent::setScissorComponent(scissor_component);
220 background_.setScissorComponent(scissor_component_);
221 text_.setScissorComponent(scissor_component_);
222 }
223 protected:
224 ButtonStyle style_;
225 Button* button_;
226 bool show_on_colors_;
227 bool primary_ui_button_;
228 bool down_;
229 bool hover_;
230 float hover_amount_;
231 OpenGlQuad background_;

237 Colour off_color_;

239 Colour off_hover_color_;
