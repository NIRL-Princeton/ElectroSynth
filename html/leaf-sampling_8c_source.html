57void tBuffer_tick (tBuffer const s, Lfloat sample)
58{
59 if (s->active == 1)
60 {
61 s->buff[s->idx] = sample;
62
63 s->idx += 1;
64
65 if (s->idx >= s->bufferLength)
66 {
67 if (s->mode == RecordOneShot)
68 {
69 tBuffer_stop(s);
70 }
71 else if (s->mode == RecordLoop)
72 {
73 s->idx = 0;
74 }
75 }
76 s->recordedLength = s->idx;
77 }

170void tSampler_initToPool(tSampler* const sp, tBuffer* const b, tMempool* const mp, LEAF* const leaf)
171{
172 _tMempool* m = *mp;
173 _tSampler* p = *sp = (_tSampler*) mpool_alloc(sizeof(_tSampler), m);
174 p->mempool = m;
175
176 _tBuffer* s = *b;
177
178 p->invSampleRate = leaf->invSampleRate;
179 p->sampleRate = leaf->sampleRate;
180 p->ticksPerSevenMs = 0.007f * p->sampleRate;
181 p->rateFactor = s->sampleRate * p->invSampleRate;
182 p->channels = s->channels;
183
184 p->samp = s;
185
186 p->active = 0;
187
188 p->start = 0;
189 p->end = p->samp->bufferLength - 1;
190
191 p->len = p->end - p->start;
192
193 p->idx = 0.f;
194 Lfloat rate = p->rateFactor; //adjust for sampling rate of buffer (may be different from leaf.sampleRate if audio file was loaded form SD card)
195 if (rate < 0.f)
196 {
197 rate = -rate;
198 p->dir = -1;
199 }
200 else
201 {
202 p->dir = 1;
203 }
204
205 p->inc = rate;
206 p->iinc = 1.f / p->inc;
207
208 p->dir = 1;
209 p->flip = 1;
210 p->bnf = 1;
211
212 p->mode = PlayNormal;
213
214 p->cfxlen = 500; // default 300 sample crossfade
215
216 tRamp_initToPool(&p->gain, 5.0f, 1, mp);
217 tRamp_setVal(p->gain, 0.f);
218
219 p->targetstart = -1;
220 p->targetend = -1;
221
222 p->inCrossfade = 0;
223 p->flipStart = -1;
224 p->flipIdx = -1;
225}

250Lfloat tSampler_tick (tSampler const p)
251{
252 attemptStartEndChange(p);
253
254 if (p->active == 0) return 0.f;
255
256 if ((p->inc == 0.0f) || (p->len < 2))
257 {
258 // p->inCrossfade = 1;
259 return p->last;
260 }
261
262 Lfloat sample = 0.0f;
263 Lfloat cfxsample = 0.0f;
264 Lfloat crossfadeMix = 0.0f;
265 Lfloat flipsample = 0.0f;
266 Lfloat flipMix = 0.0f;
267
268 Lfloat* buff = p->samp->buff;
269
270 // Variables so start is also before end
271 int myStart = p->start;
272 int myEnd = p->end;
273 if (p->flip < 0)
274 {
275 myStart = p->end;
276 myEnd = p->start;
277 }
278
279 // Get the direction and a reverse flag for some calcs
280 int dir = p->bnf * p->dir * p->flip;
281 int rev = 0;
282 if (dir < 0) rev = 1;
283
284 // Get the current integer index and alpha for interpolation
285 int idx = (int) p->idx;
286 Lfloat alpha = rev + (p->idx - idx) * dir;
287 idx += rev;
288
289 // Get the indexes for interpolation
290 int i1 = idx-(1*dir);
291 int i2 = idx;
292 int i3 = idx+(1*dir);
293 int i4 = idx+(2*dir);
294
295 int length = p->samp->recordedLength;
296
297 // Wrap as needed
298 i1 = (i1 < length*rev) ? i1 + (length * (1-rev)) : i1 - (length * rev);
299 i2 = (i2 < length*rev) ? i2 + (length * (1-rev)) : i2 - (length * rev);
300 i3 = (i3 < length*(1-rev)) ? i3 + (length * rev) : i3 - (length * (1-rev));
301 i4 = (i4 < length*(1-rev)) ? i4 + (length * rev) : i4 - (length * (1-rev));
302
303 sample = LEAF_interpolate_hermite_x (buff[i1],
304 buff[i2],
305 buff[i3],
306 buff[i4],
307 alpha);
308
309 int32_t cfxlen = p->cfxlen;
310 if (p->len * 0.25f < cfxlen) cfxlen = p->len * 0.25f;
311
312 // Determine crossfade points
313 int32_t fadeLeftStart = 0;
314 if (myStart >= cfxlen) fadeLeftStart = myStart - cfxlen;
315 int32_t fadeLeftEnd = fadeLeftStart + cfxlen;
316
317 int32_t fadeRightEnd = myEnd;// + (fadeLeftEnd - start);
318 // if (fadeRightEnd >= length) fadeRightEnd = length - 1;
319 int32_t fadeRightStart = fadeRightEnd - cfxlen;
320
321 if (p->mode == PlayLoop)
322 {
323
324 int offset = 0;
325 int cdx = 0;
326 if ((fadeLeftStart <= idx) && (idx <= fadeLeftEnd))
327 {
328 offset = fadeLeftEnd - idx;
329 cdx = fadeRightEnd - offset;
330 p->inCrossfade = 1;
331 }
332 else if ((fadeRightStart <= idx) && (idx <= fadeRightEnd))
333 {
334 offset = idx - fadeRightStart;
335 cdx = fadeLeftStart + offset;
336 p->inCrossfade = 1;
337 }
338 else p->inCrossfade = 0;
339
340 if (p->inCrossfade)
341 {
342 int c1 = cdx-(1*dir);
343 int c2 = cdx;
344 int c3 = cdx+(1*dir);
345 int c4 = cdx+(2*dir);
346
347 // Wrap as needed
348 c1 = (c1 < length * rev) ? c1 + (length * (1-rev)) : c1 - (length * rev);
349 c2 = (c2 < length * rev) ? c2 + (length * (1-rev)) : c2 - (length * rev);
350 c3 = (c3 < length * (1-rev)) ? c3 + (length * rev) : c3 - (length * (1-rev));
351 c4 = (c4 < length * (1-rev)) ? c4 + (length * rev) : c4 - (length * (1-rev));
352
353 cfxsample = LEAF_interpolate_hermite_x (buff[c1],
354 buff[c2],
355 buff[c3],
356 buff[c4],
357 alpha);
358 if (cfxlen > 0.0f) crossfadeMix = (Lfloat) offset / (Lfloat) cfxlen;
359 else crossfadeMix = 0.0f;
360 }
361
362 Lfloat flipLength = fabsf(p->flipIdx - p->flipStart);
363 if (flipLength > cfxlen)
364 {
365 p->flipStart = -1;
366 p->flipIdx = -1;
367 }
368 if (p->flipIdx >= 0)
369 {
370 if (p->flipStart == -1)
371 {
372 p->flipStart = p->idx;
373 p->flipIdx = p->idx;
374 }
375 flipLength = fabsf(p->flipIdx - p->flipStart);
376
377 int fdx = (int) p->flipIdx;
378 Lfloat falpha = (1-rev) - (p->flipIdx - fdx) * dir;
379 idx += (1-rev);
380
381 // Get the indexes for interpolation
382 int f1 = fdx+(1*dir);
383 int f2 = fdx;
384 int f3 = fdx-(1*dir);
385 int f4 = fdx-(2*dir);
386
387 // Wrap as needed
388 f1 = (f1 < length*(1-rev)) ? f1 + (length * rev) : f1 - (length * (1-rev));
389 f2 = (f2 < length*(1-rev)) ? f2 + (length * rev) : f2 - (length * (1-rev));
390 f3 = (f3 < length*rev) ? f3 + (length * (1-rev)) : f3 - (length * rev);
391 f4 = (f4 < length*rev) ? f4 + (length * (1-rev)) : f4 - (length * rev);
392
393 flipsample = LEAF_interpolate_hermite_x (buff[f1],
394 buff[f2],
395 buff[f3],
396 buff[f4],
397 falpha);
398 flipMix = (Lfloat) (cfxlen - flipLength) / (Lfloat) cfxlen;
399 }
400 }
401
402 Lfloat inc = fmodf(p->inc, (Lfloat)p->len);
403 p->idx += (dir * inc);
404 if (p->flipStart >= 0)
405 {
406 p->flipIdx += (-dir * inc);
407 if((int)p->flipIdx < 0)
408 {
409 p->idx += (Lfloat)length;
410 }
411 if((int)p->idx >= length)
412 {
413
414 p->idx -= (Lfloat)length;
415 }
416 }
417
418
419 attemptStartEndChange(p);
420
421
422 if (p->mode == PlayLoop)
423 {
424 if((int)p->idx < myStart)
425 {
426 p->idx += (Lfloat)(fadeRightEnd - fadeLeftEnd);
427 }
428 if((int)p->idx > myEnd)
429 {
430
431 p->idx -= (Lfloat)(fadeRightEnd - fadeLeftEnd);
432 }
433 }
434 else if (p->mode == PlayBackAndForth)
435 {
436 if (p->idx < myStart)
437 {
438 p->bnf = -p->bnf;
439 p->idx = myStart + 1;
440 }
441 else if (p->idx > myEnd)
442 {
443 p->bnf = -p->bnf;
444 p->idx = myEnd - 1;
445 }
446 }
447
448
449 if (p->mode == PlayNormal)
450 {
451 if (p->idx < myStart)
452 {
453 p->idx = myStart;
454 }
455 else if (p->idx > myEnd)
456 {
457 p->idx = myEnd;
458 }
459 Lfloat ticksToEnd = rev ? ((idx - myStart) * p->iinc) : ((myEnd - idx) * p->iinc);
460 if ((ticksToEnd < p->ticksPerSevenMs) && (p->active == 1))
461 {
462 tRamp_setDest(p->gain, 0.f);
463 p->active = -1;
464 }
465 }
466
467 sample = ((sample * (1.0f - crossfadeMix)) + (cfxsample * crossfadeMix)) * (1.0f - flipMix) + (flipsample * flipMix);
468
469 sample = sample * tRamp_tick(p->gain);
470
471 if (p->active < 0)
472 {
473 if (tRamp_sample(p->gain) <= 0.00001f)
474 {
475 if (p->retrigger == 1)
476 {
477 p->active = 1;
478 p->retrigger = 0;
479 tRamp_setDest(p->gain, 1.f);
480
481 if (p->dir > 0)
482 {
483 if (p->flip > 0) p->idx = p->start;
484 else p->idx = p->end;
485 }
486 else
487 {
488 if (p->flip > 0) p->idx = p->end;
489 else p->idx = p->start;
490 }
491 }
492 else
493 {
494 p->active = 0;
495 }
496
497 }
498 }
499
500 p->last = sample;
501
502
503 return p->last;
504}
