27void tPRCReverb_initToPool (tPRCReverb* const rev, Lfloat t60, tMempool* const mp)
28{
29 _tMempool* m = *mp;
30 _tPRCReverb* r = *rev = (_tPRCReverb*) mpool_alloc(sizeof(_tPRCReverb), m);
31 r->mempool = m;
32 LEAF* leaf = r->mempool->leaf;
33
34 if (t60 <= 0.0f) t60 = 0.001f;
35
36 r->sampleRate = leaf->sampleRate;
37 r->invSampleRate = leaf->invSampleRate;
38
39 int lengths[4] = { 341, 613, 1557, 2137 }; // Delay lengths for 44100 Hz sample rate.
40 double scaler = r->sampleRate * INV_44100;
41
42 int delay, i;
43 if (scaler != 1.0f)
44 {
45 for (i=0; i<4; i++)
46 {
47 delay = (int) scaler * lengths[i];
48
49 if ( (delay & 1) == 0) delay++;
50
51 while ( !LEAF_isPrime(delay) ) delay += 2;
52
53 lengths[i] = delay;
54 }
55 }
56
57 tDelay_initToPool(&r->allpassDelays[0], lengths[0], lengths[0] * 2, mp);
58 tDelay_initToPool(&r->allpassDelays[1], lengths[1], lengths[1] * 2, mp);
59 tDelay_initToPool(&r->combDelay, lengths[2], lengths[2] * 2, mp);
60
61 tPRCReverb_setT60(*rev, t60);
62
63 r->allpassCoeff = 0.7f;
64 r->mix = 0.5f;
65}

98Lfloat tPRCReverb_tick(tPRCReverb const r, Lfloat input)
99{
100 Lfloat temp, temp0, temp1, temp2;
101 Lfloat out;
102
103 r->lastIn = input;
104
105 temp = tDelay_getLastOut(r->allpassDelays[0]);
106 temp0 = r->allpassCoeff * temp;
107 temp0 += input;
108 tDelay_tick(r->allpassDelays[0], temp0);
109 temp0 = -( r->allpassCoeff * temp0) + temp;
110
111 temp = tDelay_getLastOut(r->allpassDelays[1]);
112 temp1 = r->allpassCoeff * temp;
113 temp1 += temp0;
114 tDelay_tick(r->allpassDelays[1], temp1);
115 temp1 = -(r->allpassCoeff * temp1) + temp;
116
117 temp2 = temp1 + ( r->combCoeff * tDelay_getLastOut(r->combDelay));
118
119 out = r->mix * tDelay_tick(r->combDelay, temp2);
120
121 temp = (1.0f - r->mix) * input;
122
123 out += temp;
124
125 r->lastOut = out;
126
127 return out;
128}

170void tNReverb_initToPool (tNReverb* const rev, Lfloat t60, tMempool* const mp)
171{
172 _tMempool* m = *mp;
173 _tNReverb* r = *rev = (_tNReverb*) mpool_alloc(sizeof(_tNReverb), m);
174 r->mempool = m;
175 LEAF* leaf = r->mempool->leaf;
176
177 if (t60 <= 0.0f) t60 = 0.001f;
178
179 r->sampleRate = leaf->sampleRate;
180 r->invSampleRate = leaf->invSampleRate;
181
182 int lengths[15] = {1433, 1601, 1867, 2053, 2251, 2399, 347, 113, 37, 59, 53, 43, 37, 29, 19}; // Delay lengths for 44100 Hz sample rate.
183 double scaler = r->sampleRate * INV_44100;// / 25641.0f;
184
185 int delay, i;
186
187 for (i=0; i < 15; i++)
188 {

194 lengths[i] = delay;
195 }
196

259Lfloat tNReverb_tick(tNReverb const r, Lfloat input)
260{
261 r->lastIn = input;
262
263 Lfloat temp, temp0, temp1, temp2, out;
264 int i;
265
266 temp0 = 0.0;
267 for ( i=0; i<6; i++ )
268 {
269 temp = input + (r->combCoeffs[i] * tLinearDelay_getLastOut(r->combDelays[i]));
270 temp0 += tLinearDelay_tick(r->combDelays[i],temp);
271 }
272
273 for ( i=0; i<3; i++ )
274 {
275 temp = tLinearDelay_getLastOut(r->allpassDelays[i]);
276 temp1 = r->allpassCoeff * temp;
277 temp1 += temp0;
278 tLinearDelay_tick(r->allpassDelays[i], temp1);
279 temp0 = -(r->allpassCoeff * temp1) + temp;
280 }
281
282 // One-pole lowpass filter.
283 r->lowpassState = 0.7f * r->lowpassState + 0.3f * temp0;
284
285 temp = tLinearDelay_getLastOut(r->allpassDelays[3]);
286 temp1 = r->allpassCoeff * temp;
287 temp1 += r->lowpassState;
288 tLinearDelay_tick(r->allpassDelays[3], temp1 );
289 temp1 = -(r->allpassCoeff * temp1) + temp;
290
291 temp = tLinearDelay_getLastOut(r->allpassDelays[4]);
292 temp2 = r->allpassCoeff * temp;
293 temp2 += temp1;
294 tLinearDelay_tick(r->allpassDelays[4], temp2 );
295 out = -( r->allpassCoeff * temp2 ) + temp ;
296
297 //the other channel in stereo version below
298/*
299 temp = tLinearDelay_getLastOut(&r->allpassDelays[5]);
300 temp3 = r->allpassCoeff * temp;
301 temp3 += temp1;
302 tLinearDelay_tick(&r->allpassDelays[5], temp3 );
303 out = r->mix *( - ( r->allpassCoeff * temp3 ) + temp );
304*/
305
306 temp = ( 1.0f - r->mix ) * input;
307
308 out += temp;
309
310 r->lastOut = out;
311
312 return out;
313}

315void tNReverb_tickStereo(tNReverb const r, Lfloat input, Lfloat* output)
316{
317 r->lastIn = input;
318
319 Lfloat temp, temp0, temp1, temp2, temp3, out;
320 int i;
321
322 temp0 = 0.0;
323 for ( i=0; i<6; i++ )
324 {
325 temp = input + (r->combCoeffs[i] * tLinearDelay_getLastOut(r->combDelays[i]));
326 temp0 += tLinearDelay_tick(r->combDelays[i],temp);
327 }
328
329 for ( i=0; i<3; i++ )
330 {
331 temp = tLinearDelay_getLastOut(r->allpassDelays[i]);
332 temp1 = r->allpassCoeff * temp;
333 temp1 += temp0;
334 tLinearDelay_tick(r->allpassDelays[i], temp1);
335 temp0 = -(r->allpassCoeff * temp1) + temp;
336 }
337
338 // One-pole lowpass filter.
339 r->lowpassState = 0.7f * r->lowpassState + 0.3f * temp0;
340
341 temp = tLinearDelay_getLastOut(r->allpassDelays[3]);
342 temp1 = r->allpassCoeff * temp;
343 temp1 += r->lowpassState;
344 tLinearDelay_tick(r->allpassDelays[3], temp1 );
345 temp1 = -(r->allpassCoeff * temp1) + temp;
346
347 Lfloat drymix = ( 1.0f - r->mix ) * input;
348
349 temp = tLinearDelay_getLastOut(r->allpassDelays[4]);
350 temp2 = r->allpassCoeff * temp;
351 temp2 += temp1;
352 tLinearDelay_tick(r->allpassDelays[4], temp2 );
353 output[0] = -( r->allpassCoeff * temp2 ) + temp + drymix;
354 out = output[0];
355
356 temp = tLinearDelay_getLastOut(r->allpassDelays[5]);
357 temp3 = r->allpassCoeff * temp;
358 temp3 += temp1;
359 tLinearDelay_tick(r->allpassDelays[5], temp3 );
360 output[1] = r->mix *( - ( r->allpassCoeff * temp3 ) + temp + drymix);
361
362 r->lastOut = out;
363}

529Lfloat tDattorroReverb_tick (tDattorroReverb const r, Lfloat input)
530{
531 Lfloat in_sample, f1_sample,f1_delay_2_sample, f2_sample, f2_delay_2_sample;
532
533 if (r->frozen)
534 {
535 input = 0.0f;
536 //r->f1_last = 0.0f;
537 //r->f2_last = 0.0f;
538 }
539 // INPUT
540 in_sample = tTapeDelay_tick(r->in_delay, input);
541
542 in_sample = tOnePole_tick(r->in_filter, in_sample);
543
544 for (int i = 0; i < 4; i++)
545 {
546 in_sample = tAllpass_tick(r->in_allpass[i], in_sample);
547 }
548
549 // FEEDBACK 1
550 f1_sample = in_sample + r->f2_last; // + f2_last_out;
551
552 tAllpass_setDelay(r->f1_allpass, SAMP(30.51f) + tCycle_tick(r->f1_lfo) * SAMP(4.0f));
553
554 f1_sample = tAllpass_tick(r->f1_allpass, f1_sample);
555
556 f1_sample = tTapeDelay_tick(r->f1_delay_1, f1_sample);
557
558 f1_sample = tOnePole_tick(r->f1_filter, f1_sample);
559
560 f1_sample = f1_sample + r->f1_delay_2_last * 0.5f;
561
562 f1_delay_2_sample = tTapeDelay_tick(r->f1_delay_2, f1_sample * 0.5f);
563
564 r->f1_delay_2_last = f1_delay_2_sample;
565
566 f1_sample = r->f1_delay_2_last + f1_sample;
567
568 f1_sample = tHighpass_tick(r->f1_hp, f1_sample);
569
570 f1_sample *= r->feedback_gain;
571
572 r->f1_last = tTapeDelay_tick(r->f1_delay_3, f1_sample);
573
574 // FEEDBACK 2
575 f2_sample = in_sample + r->f1_last;
576
577 tAllpass_setDelay(r->f2_allpass, SAMP(22.58f) + tCycle_tick(r->f2_lfo) * SAMP(4.0f));
578
579 f2_sample = tAllpass_tick(r->f2_allpass, f2_sample);
580
581 f2_sample = tTapeDelay_tick(r->f2_delay_1, f2_sample);
582
583 f2_sample = tOnePole_tick(r->f2_filter, f2_sample);
584
585 f2_sample = f2_sample + r->f2_delay_2_last * 0.5f;
586
587 f2_delay_2_sample = tTapeDelay_tick(r->f2_delay_2, f2_sample * 0.5f);
588
589 r->f2_delay_2_last = f2_delay_2_sample;
590
591 f2_sample = r->f2_delay_2_last + f2_sample;
592
593 f2_sample = tHighpass_tick(r->f2_hp, f2_sample);
594
595 f2_sample *= r->feedback_gain;
596
597 r->f2_last = tTapeDelay_tick(r->f2_delay_3, f2_sample);
598
599 // TAP OUT 1
600 f1_sample = tTapeDelay_tapOut(r->f1_delay_1, SAMP(8.9f)) +
601 tTapeDelay_tapOut(r->f1_delay_1, SAMP(99.8f));
602
603 f1_sample -= tTapeDelay_tapOut(r->f1_delay_2, SAMP(64.2f));
604
605 f1_sample += tTapeDelay_tapOut(r->f1_delay_3, SAMP(67.f));
606
607 f1_sample -= tTapeDelay_tapOut(r->f2_delay_1, SAMP(66.8f));
608
609 f1_sample -= tTapeDelay_tapOut(r->f2_delay_2, SAMP(6.3f));
610
