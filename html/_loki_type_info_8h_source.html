28// class TypeInfo
29// Purpose: offer a first-class, comparable wrapper over std::type_info
31
32 class TypeInfo
33 {
34 public:
35 // Constructors
36 TypeInfo(); // needed for containers
37 TypeInfo(const std::type_info&); // non-explicit
38
39 // Access for the wrapped std::type_info
40 const std::type_info& Get() const;
41 // Compatibility functions
42 bool before(const TypeInfo& rhs) const;
43 const char* name() const;
44
45 private:
46 const std::type_info* pInfo_;
47 };
48
49// Implementation
50
51 inline TypeInfo::TypeInfo()
52 {
53 class Nil {};
54 pInfo_ = &typeid(Nil);
55 assert(pInfo_);
56 }
57
58 inline TypeInfo::TypeInfo(const std::type_info& ti)
59 : pInfo_(&ti)
60 { assert(pInfo_); }
61
62 inline bool TypeInfo::before(const TypeInfo& rhs) const
63 {
64 assert(pInfo_);
65 // type_info::before return type is int in some VC libraries
66 return pInfo_->before(*rhs.pInfo_) != 0;
67 }
68
69 inline const std::type_info& TypeInfo::Get() const
70 {
71 assert(pInfo_);
72 return *pInfo_;
73 }
74
75 inline const char* TypeInfo::name() const
76 {
77 assert(pInfo_);
78 return pInfo_->name();
79 }
80
81// Comparison operators
82
83 inline bool operator==(const TypeInfo& lhs, const TypeInfo& rhs)
84 // type_info::operator== return type is int in some VC libraries
85 { return (lhs.Get() == rhs.Get()) != 0; }
86
87 inline bool operator<(const TypeInfo& lhs, const TypeInfo& rhs)
88 { return lhs.before(rhs); }
89
90 inline bool operator!=(const TypeInfo& lhs, const TypeInfo& rhs)
91 { return !(lhs == rhs); }
92
93 inline bool operator>(const TypeInfo& lhs, const TypeInfo& rhs)
94 { return rhs < lhs; }
95
96 inline bool operator<=(const TypeInfo& lhs, const TypeInfo& rhs)
97 { return !(lhs > rhs); }
98
99 inline bool operator>=(const TypeInfo& lhs, const TypeInfo& rhs)
100 { return !(lhs < rhs); }
101}
