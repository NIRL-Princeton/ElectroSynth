111 void reserve(int capacity) {
112 int new_capacity = capacity + 1;
113 if (new_capacity < capacity_)
114 return;
115
116 std::unique_ptr<T[]> tmp = std::make_unique<T[]>(new_capacity);
117
118 if (capacity_) {
119 end_ = size();
120 for (int i = 0; i < end_; ++i)
121 tmp[i] = std::move(at(i));
122 }
123
124 data_ = std::move(tmp);
125 capacity_ = new_capacity;
126 start_ = 0;
127 }

129 force_inline void assign(int num, T value) {

131 reserve(num);

134 data_[i] = value;

137 end_ = num;
138 }

178 force_inline void removeAt(int index) {
179 _ASSERT(end_ != start_);
180 int i = (index + start_) % capacity_;
181 end_ = (end_ - 1 + capacity_) % capacity_;
182 while (i != end_) {

186 }

207 template<int(*compare)(T, T)>
208 void sort() {
209 int total = size();
210 for (int i = 1; i < total; ++i) {
211 T value = at(i);
212 int j = i - 1;
213 for (; j >= 0 && compare(at(j), value) < 0; --j)
214 at(j + 1) = at(j);
215
216 at(j + 1) = value;
217 }
218 }
